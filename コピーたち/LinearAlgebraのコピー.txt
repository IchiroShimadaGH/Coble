SubMatrix:=function(A, rowrange, colrange)
	local ii, jj, v, sA, newv;
	sA:=[];
	for ii in rowrange do
		newv:=[];
		v:=A[ii];
		for jj in colrange do
			Add(newv, v[jj]);
		od;
		Add(sA, newv);
	od;
	return(sA);
end;

GcdTransform:=function(as)
	#
	# as is a list of integers of length k.
	# d is the gcd of as.
	# gcdtransrec.gcd is d.
	# gcdtransrec.eulcidv is a vector v such that v*as=d.
	# gcdtransrec.nullbasis is a basis of the Z-module
	# (in HermiteNormalForm) consisting of u's such that u*a=0.
	#
	local beep, k, d, tas, snf, gcdtransrec, ev, kk, nullbasis;
	#
	beep:=function(n)
		Printn("wrong GcdTransform", n); Error();
	end;
	#
	k:=Length(as);
	d:=Gcd(as);
	if d=0 then beep(1); fi;
	tas:=TransposedMat([as]);
	snf:=SmithNormalFormIntegerMatTransforms(tas);
	if snf.normal[1][1]<>d then beep(2); fi;
	if snf.coltrans=[[1]] then
		ev:=snf.rowtrans[1];
	elif snf.coltrans=[[-1]] then
		ev:=(-1)*(snf.rowtrans[1]);
	else
		beep(3);
	fi;
	nullbasis:=HermiteNormalFormIntegerMat(List([2..k], kk->snf.rowtrans[kk]));
	#
	gcdtransrec:=rec();
	gcdtransrec.gcd:=d;
	gcdtransrec.euclidv:=ev;
	gcdtransrec.nullbasis:=nullbasis;
	return(gcdtransrec);
end;



FindIndependentVects:=function(vects, targetrank)
	#
	#
	local fflag, partial, findbasisjob, basisiis;
	#
	if RankMat(vects)<targetrank then
		Printn("wrong input in  FindIndependentVects");
		Error();
	fi;
	#
	fflag:=false;
	partial:=MakeZeroVect(targetrank); #holder
	#
	findbasisjob:=function(level, partialvects, lastentry)
		local kk, newpartialvects;
		if level>targetrank then
			basisiis:=List(partial);
			fflag:=true;
			return();
		fi;
		for kk in [lastentry+1..Length(vects)] do
			newpartialvects:=CopyAdd(partialvects, vects[kk]);
			if RankMat(newpartialvects)=level then
				partial[level]:=kk;
				findbasisjob(level+1, newpartialvects, kk);
				if fflag then break; fi; # from for kk in
			fi;
		od;
	end;
	findbasisjob(1, [], 0);
	if not fflag then
		Printn("wrong  FindIndependentVects"); Error();
	fi;
	return(basisiis);
end;

HermiteNormalFormIntMatModN:=function(intmat, N)
	local dim, r, ii, tM, tHM, newintmat, vv, vvmodN;

	r:=Length(intmat);
	tM:=StackMats(intmat, N*IdentityMat(dim));

	newintmat:=[];
	for vv in tHM do

		fi;

end;


OrthogonalModN:=function(vs, N)
	#
	# Z-basis (in HermiteNormalForm) of the
	# Z-modles consisting of x such that
	# v*x mod N=0 for all v in vs.
	#
	local tvs, dim, hvs, snf, ii, gens, zs, dii;
	dim:=Length(vs[1]);
	hvs:=HermiteNormalFormIntMatModN(vs, N);
	if IsZeroMat(hvs) then return(IdentityMat(dim)); fi;
	tvs:=TransposedMat(hvs);
	snf:=SmithNormalFormIntegerMatTransforms(tvs);
	gens:=[];
	for ii in [1..Length(hvs)] do
		dii:=snf.normal[ii][ii];
		Add(gens, Lcm(dii, N)/dii*MakeVectei(dim, ii));
	od;
	for ii in [Length(hvs)+1..dim] do
		Add(gens, MakeVectei(dim, ii));
	od;
	zs:=gens*snf.rowtrans;
	zs:=HermiteNormalFormIntMatModN(zs, N);
	return(zs);
end;

AbSolModN:=function(A, b, N)
	#
	# solve the equation A*x=b over Z/NZ.
	# [s, ms].
	# s satisfies A*s=b mod N.
	# ms is a set of generators of the module
	# defined by A*x=0.
	#
	local sm, Ub, V, D, s, ms, ii, d, jj, sii, dim, g,
	dprime, Nprime, Ubprime, dprimeinv;
	#
	if Length(A)<>Length(b) then Printn("wrong AbSolModN"); beep(1); fi;
	dim:=ColdimMat(A);
	sm:=SmithNormalFormIntegerMatTransforms(A mod N);
	Ub:=((sm.rowtrans)*b) mod N;
	D:=RemoveZeroVectsIntMat(sm.normal mod N);
	for jj in [Length(D)+1..Length(Ub)] do
		if Ub[jj]<>0 then return(fail); fi;
	od;
	s:=[];
	ms:=[];
	for ii in [1..Length(D)] do
		d:=D[ii][ii];
		g:=GcdInt(d, N);
		if Ub[ii] mod g <>0 then return(fail); fi;
		dprime:=d/g;
		Nprime:=N/g;
		Ubprime:=Ub[ii]/g;
		if Nprime<>N then
			Add(ms, Nprime*MakeVectei(dim, ii));
		fi;
		dprimeinv:=(1/dprime) mod Nprime;
		Add(s, dprimeinv*Ubprime mod N);
	od;
	for ii in [Length(D)+1..dim] do
		Add(s, 0);
		Add(ms, MakeVectei(dim, ii));
	od;
	V:=sm.coltrans mod N;
	s:=(V*s) mod N;
	ms:=List(ms, vv->(V*vv) mod N);
	return([s, ms]);
end;

FindRankM:=function(listv, rank)
	local basis, lastii, ii, vii;
	#
	if Rank(listv)<rank then
		Printn("wrong input in FindRankM"); Error();
	fi;
	lastii:=0;
	basis:=[];
	while Length(basis)<rank do
		for ii in [lastii+1..Length(listv)] do
			vii:=listv[ii];
			Add(basis, vii);
			if Rank(basis)=Length(basis) then
				lastii:=ii;
				break; #from for ii in
			else
				Remove(basis);
			fi;
		od;
	od;
	if Length(basis)<>rank then
		Printn("wrong in FindRankM"); Error();
	fi;
	return(basis);
end;

RestrictionToSub:=function(emb, gL)
	#
	# gL : L \to L preserves the image of emb : M\inj L.
	# find gM : M \to M  such that emb*gL=gM*emb.
	#
	local T1, T2, h1rec, h2rec, gM;
	#
	h1rec:=HermiteNormalFormIntegerMatTransform(emb);
	h2rec:=HermiteNormalFormIntegerMatTransform(emb*gL);
	if h1rec.normal<>h2rec.normal then
		Printn("wrong RestrictionToSub", 1); Error();
	fi;
	T1:=h1rec.rowtrans;
	T2:=h2rec.rowtrans;
	gM:=InverseMat(T2)*T1;
	if emb*gL<>gM*emb then
		Printn("wrong RestrictionToSub", 2); Error();
	fi;
	return(gM);
end;

MXB:=function(M, B)
	# the row vectors of B are linearly independent.
	# the row vectors of M are in <B> tensor Q.
	# M=X*B.
	# the idea is as follows.
	# S*B*T=[D|O] (SmithNormalForm)
	# B*T=[Sinv*D|O]
	# M*T=X*B*T=[X*Sinv*D|O]
	#
	local beep, X, n, r, smnf, S, T, D, MT, lengthM, MT1, MT2;
	#
	beep:=function(beepnumb)
		Printn("wrong in MXB", beepnumb); Error();
	end;
	#
	n:=Length(B[1]);
	r:=Rank(B);
	if r<>Length(B) then beep(45344); fi;
	smnf:=SmithNormalFormIntegerMatTransforms(B);
	S:=smnf.rowtrans;
	T:=smnf.coltrans;
	D:=SubMatrix(smnf.normal, [1..r], [1..r]);
	MT:=M*T;
	lengthM:=Length(M);
	MT1:=SubMatrix(MT, [1..lengthM], [1..r]);
	MT2:=SubMatrix(MT, [1..lengthM], [r+1..n]);
	if not IsZeroMat(MT2) then
		beep(5456774);
		#return("<M> tensor Q is not in <B> tensor Q");
	fi;
	X:=MT1*InverseMat(D)*S;
	if M<>X*B then
		beep(42785238923);
	fi;
	return(X);
end;

AXisB:=function(A, B)
	#
	# A is of full rank.
	# Find X such that AX=B.
	#
	local beep, n, hnf, SBupsub, Dbarinv, tg, SB;
	#
	beep:=function(beepnumb)
		localbeep("AXisB", beepnumb);
	end;
	#
	n:=Length(A[1]);
	if Rank(A)<>n then Printn("AXisB: Rank(A) is wrong"); beep(281828); fi;
	hnf:=HermiteNormalFormIntegerMatTransforms(A);
	#
	SB:=hnf.rowtrans*B;
	SBupsub:=List([1..n], ii->SB[ii]);
	Dbarinv:=InverseMat(List([1..n], ii->hnf.normal[ii]));
	#
	tg:=Dbarinv*SBupsub;
	if A*tg<>B then Printn("AXisB: no solution"); beep(916371); fi;
	#
	return(tg);
end;

	# Find X such that AX=B.
	#
	local beep, n, hnf, SBupsub, Dbarinv, tg, SB;
	#
	beep:=function(beepnumb)
		localbeep("AXisB", beepnumb);
	end;
	#
	n:=Length(A[1]);
	if Rank(A)<>n then Printn("AXisB: Rank(A) is wrong"); beep(281828); fi;
	hnf:=HermiteNormalFormIntegerMatTransforms(A);
	#
	SB:=hnf.rowtrans*B;
	SBupsub:=List([1..n], ii->SB[ii]);
	Dbarinv:=InverseMat(List([1..n], ii->hnf.normal[ii]));
	#
	tg:=Dbarinv*SBupsub;
	if A*tg<>B then return(fail); fi;
	#
	return(tg);
end;

RzvHnf:=function(M)
  local hnf, nznops, newhnf, ii, beep;
  #
  beep:=function(beepnumb)
    localbeep("RzvHnf", beepnumb);Error();
  end;
  #
  if IsZeroMat(M) then
    Printn("RzvHnf: input is a zero matrix"); beep(896262987);
  fi;
  hnf:=HermiteNormalFormIntegerMatTransform(M);
  nznops:=Length(M)-OccurNumb(List(hnf.normal, IsZeroVect), true);
  newhnf:=rec(
    normal:=List([1..nznops], ii->hnf.normal[ii]),
    rowtrans:=List([1..nznops], ii->hnf.rowtrans[ii])
  );
  return(newhnf);
end;


NullDecompAoplusB:=function(gensA, gensB)
  #
  local Id, tn, beep, nullA, nullB, decomprec;
  #
  beep:=function(beepnumb)
    localbeep("NullDecompAoplusB", beepnumb);Error();
  end;
  #
  nullA:=(gensA=[] or Rank(gensA)=0);
  nullB:=(gensB=[] or Rank(gensB)=0);
  if nullA and nullB then
    Printn("wrong input", gensA, gensB);beep(1331);
  fi;
  if nullA then
    tn:=Rank(gensB);
    Id:=IdentityMat(tn);
    decomprec:=rec(
      embA:=[],
      embB:=Id,
      projA:=[],
      projB:=Id
    );
    return(decomprec);
  elif nullB then
    tn:=Rank(gensA);
    Id:=IdentityMat(tn);
    decomprec:=rec(
      embA:=Id,
      embB:=[],
      projA:=Id,
      projB:=[]
    );
    return(decomprec);
  else
    beep(63938);
  fi;
end;

DecompAoplusB:=function(gensA, gensB)
  local beep, decomprec, embA, embB, projA, projB,
  totalrank, rankA, rankB, embs, embsinv;
  #
  beep:=function(beepnumb)
    localbeep("DecompAoplusB", beepnumb);Error();
  end;
  #
  if gensA=[] or gensB=[] or Rank(gensA)=0 or Rank(gensB)=0 then
    return(NullDecompAoplusB(gensA, gensB));
  fi;
  rankA:=Rank(gensA);
  rankB:=Rank(gensB);
  totalrank:=Rank(Concatenation(gensA, gensB));
  if rankA+rankB<>totalrank then
    beep(6296982);
  fi;
  #
  embA:=RzvHnf(gensA).normal;
  embB:=RzvHnf(gensB).normal;
  embs:=StackMats(gensA, gensB);
  embsinv:=InverseMat(embs);
  projA:=SubMatrix(embsinv, [1..totalrank], [1..rankA]);
  projB:=SubMatrix(embsinv, [1..totalrank], [rankA+1..totalrank]);
  #
  decomprec:=rec(
    embA:=embA,
    embB:=embB,
    projA:=projA,
    projB:=projB
  );
  return(decomprec);
end;

MakePrimitiveIntMat:=function(M)
  local beep, v,dd,nn,aa,ss,primM, intmatrec, ssv;
  #
  beep:=function(beepnumb)
    localbeep("MakePrimitiveIntMat", beepnumb); Error();
  end;
  #
  v:=Flat(M);
  dd:=Llcm(List(v, DenominatorRat));
  nn:=Lgcd(List(dd*v, NumeratorRat));
  ss:=dd/nn;
  ssv:=ss*v;
  if not IsIntVect(ssv) then beep(29234); fi;
  if Lgcd(ssv)<>1 then beep(19234); fi;
  if ss<0 then beep(69283); fi;
  primM:=ss*M;
  intmatrec:=rec(
    ssM:=primM,
    ss:=ss
  );
  return(intmatrec);
end;
