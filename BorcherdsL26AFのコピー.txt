Make_nRvRsRec:=function(embSXLdata)
	#
	# make the record for vectors in GramRdual
  # with -2<=squarenorm <=0
  # and their embeddings in L26
	#
	local beep, svrec, nRs, vRss, vv, vvnR,   nRvRsrec, pos, emb2_dual,
  vRInLss, vRs, GramL, tGramRdual, LGrec, dd;
  #
  beep:=function(beepnumb)
    localbeep("Make_nRvRsRec", beepnumb); Error();
  end;
  #
  LGrec:=embSXLdata.LGrec;
	#embSXLdata.LGrec will be made in BorcherdsL26AF
  GramL:=LGrec.Gram;
  tGramRdual:=LGrec.Gram2dual;
  emb2_dual:=LGrec.emb2_dual;
	# from the dual basis of Rdual to the STANDARD basis of L26
  dd:=Lgcd(List(Flat(tGramRdual), DenominatorRat));
  #
	svrec:=ShortestVectors((-dd)*tGramRdual, 2*dd);
	if svrec.vectors=[] then beep(72727272); fi;
  #
  nRs:=(-1)*Set(svrec.norms);
  Add(nRs, 0);
	#
  vRss:=List(nRs, nR->[]);
  Add(vRss[SinglePosition(nRs, 0)], MakeZeroVect(Length(tGramRdual)));
  for vv in svrec.vectors do
    vvnR:=vv*tGramRdual*vv;
    pos:=SinglePosition(nRs, vvnR);
    Append(vRss[pos], [vv, (-1)*vv]);
  od;
  #
  vRInLss:=List(vRss, vRs->vRs*emb2_dual);
  #
	#for check
	#
  if List(vRss, vRs->Set(vRs, vv->vv*tGramRdual*vv))<>List(nRs, xx->[xx]) then
    beep(52527);
  fi;
  if false in List(vRss, IsDuplicateFree) then beep(16191); fi;
  if false in List(vRInLss, IsDuplicateFree) then beep(126191); fi;
  if List(vRInLss, vRs->Set(vRs, vv->vv*GramL*vv))<>List(nRs, xx->[xx]) then
    beep(52527);
  fi;
  #
  nRvRsrec:=rec(
    nRs:=nRs,
    vRss:=vRss,
    vRInLss:=vRInLss
  );
	return(nRvRsrec);
end; #the end of Make_nRvRsRec:


Make_nSaSvSss:=function(embSXLdata, nowweyl)
	#
	# nSaSvSss is a list of records
	# nSaSvSsrec:=rec(nS, aS, vSliftss).
	# vSliftss is a list of [vS, lifts].
	#
  local beep, GramL, GramS, GramSdual, dd, intGramSdual, nowweyldual,
  wS, wSdual, embSdual, nSaSvSss, Rpos, nR, nS,
  aRvRInLss, aR, vRInLs, aS, lifts, task, lift, vSliftss, vSlifts, vS,
  nSaSvSsrec, aRvRInLs, nSaSs, xx, LGrec, nRvRsrec, totalvRInLs;
  #
  beep:=function(beepnumb)
    localbeep("Make_nSaSvSss", beepnumb); Error();
  end;
  #
  nRvRsrec:=embSXLdata.nRvRsrec; # will be made in BorchardsL26AF
  LGrec:=embSXLdata.LGrec;
  GramL:=LGrec.Gram;
  GramS:=LGrec.Gram1;
  GramSdual:=LGrec.Gram1dual;
  dd:=Lgcd(List(Flat(GramSdual), DenominatorRat));
  intGramSdual:=dd*GramSdual;
  nowweyldual:=nowweyl*GramL;
  wS:=nowweyl*LGrec.proj1;
  if wS*GramS*wS<=0 then beep(619611); fi;
  wSdual:=nowweyl*LGrec.proj1_dual;
  if wSdual<>wS*GramS then beep(158756781); fi;
  if not IsIntVect(wSdual) then beep(58111); fi;
  #
  embSdual:=LGrec.emb1_dual;
	#in terms of the STANDARD basis of L and the dual basis of L1
  #
  nSaSvSss:=[];
  #
  for Rpos in [1..Length(nRvRsrec.nRs)] do
    nR:=nRvRsrec.nRs[Rpos];
    if nR=-2 then continue; fi;
    nS:=-2-nR;
    if not nS<0 then beep(69816981); fi;
    totalvRInLs:=nRvRsrec.vRInLss[Rpos];
    aRvRInLss:=NewClassifyByTags(totalvRInLs, vv->vv*nowweyldual);
    #
    for aRvRInLs in aRvRInLss do
      aR:=aRvRInLs[1];
      vRInLs:=aRvRInLs[2];
      aS:=1-aR;
      #
      lifts:=[];
      #
      task:=function(vInSdual)
        local vSInL, vRInL, vL;
        vSInL:=vInSdual*embSdual;
        for vRInL in vRInLs do
          vL:=vSInL+vRInL;
          if IsIntVect(vL) then
            Add(lifts, vL);
          fi;
        od;
      end;
      #
      AffES(intGramSdual, wSdual, dd*aS, dd*nS,  true, task);
      vSliftss:=NewClassifyByTags(lifts, vv->vv*LGrec.proj1);
      for vSlifts in vSliftss do # for check
        vS:=vSlifts[1];
        if vS*GramS*vS<>nS then beep(62191); fi;
        if vS*wSdual<>aS then beep(62291); fi;
        for lift in vSlifts[2] do
          if lift*nowweyldual<>1 then beep(261112); fi;
          if lift*GramL*lift<>-2 then beep(91277); fi;
        od;
      od;
      #
      if vSliftss<>[] then
        nSaSvSsrec:=rec(
          nS:=nS,
          aS:=aS,
          vSliftss:=vSliftss
        );
        Add(nSaSvSss, nSaSvSsrec);
        #Printn("nS", nS, "aS", aS, "nops", Length(vSliftss));
      fi;
    od;
  od; #the end of for Rpos in [1..Length(nRvRsrec.nRs)] do
  #
  nSaSs:=List(nSaSvSss, xx->[xx.nS, xx.aS]);
  SortParallel(nSaSs, nSaSvSss);
  return(nSaSvSss);
end; #the end of Make_nSaSvSss.


MakeAdjRec:=function(embSXLdata, nowweyl, walllf) #lf in walllf is "linear form"
  #
  local beep, LGrec, nRvRsrec, tg, GramL, GramS, GramSdual, embSdual, nowweyldual,vRInL,
  rSdualprim, primnS, passingLwalls, kk, kkprinmnS, kkrSdualprim,
  nR, vRInLs, vL, ss, perturbdegs, kkprimnS, tgS, reflmatL, id26,projLwalls,
  kkrSprimInL,aa, embS, sqrtaa, sqrtaarSprim, adjrec, rSprim, newweyl, projS,
  newweyldual, rv;
  #
  beep:=function(beepnumb)
    localbeep("MakeAdjRec", beepnumb); Error();
  end;
  #
  nRvRsrec:=embSXLdata.nRvRsrec;
  LGrec:=embSXLdata.LGrec;
  GramL:=LGrec.Gram;
  GramS:=LGrec.Gram1;
  GramSdual:=LGrec.Gram1dual;
  embSdual:=LGrec.emb1_dual;
  # in terms of the STANDARD basis of L26 to the dual basis of Sdual
  embS:=LGrec.emb1;
	# in terms of the  basis of S to the standard basis of L26
  projS:=LGrec.proj1;
	# in terms of the  basis of S to the standard basis of L26
  nowweyldual:=nowweyl*GramL;
  if GetPrimitiveIntVect(walllf)<>walllf then beep(17737); fi;
  #
  rSdualprim:=List(walllf);
  rSprim:=rSdualprim*GramSdual;
  #
  adjrec:=rec(
    backto:=false, #Caution:this may be changed in NewMakeChamRec
    enteringwall:=rSprim
  );
  #
  primnS:=rSdualprim*GramSdual*rSdualprim;
  if not primnS<0 then beep(188127); fi;
  aa:=-2/primnS;
  sqrtaa:=SqRoot(aa); #in gaptools.txt
  if IsInt(sqrtaa) then
    sqrtaarSprim:=sqrtaa*rSprim;
    if sqrtaarSprim*GramS*sqrtaarSprim<>-2 then beep(19192); fi;
    if IsIntVect(sqrtaarSprim) then
      adjrec.israt:=true;
      adjrec.minus2vector:=sqrtaarSprim;
      return(adjrec); #this wall is a (-2)-wall
    fi;
  fi;
  #
  adjrec.israt:=false;
  passingLwalls:=[];
  kk:=0;
  kkprimnS:=0;
  kkrSdualprim:=MakeZeroVect(Length(GramS));
  while kkprimnS>=-2 do # kk:=kk+1; kkprimnS:=kk^2*primnS;
    nR:=-2-kkprimnS;
    if nR in nRvRsrec.nRs then
      kkrSprimInL:=kkrSdualprim*embSdual;
      vRInLs:=nRvRsrec.vRInLss[SinglePosition(nRvRsrec.nRs, nR)];
      for vRInL in vRInLs do
        vL:=kkrSprimInL+vRInL;
        if vL*GramL*vL<>-2 then beep(1881181); fi;
        if IsIntVect(vL) then
          if kkprimnS=-2 then beep(6916817); fi;
          #(rS)^perp should NOT be ((-2)-vect)^perp, because israt is false.
          ss:=nowweyldual*vL;
          if ss>0 then
            Add(passingLwalls, vL);
          elif ss<0 then
            Add(passingLwalls, (-1)*vL);
          else beep(616191);
          fi;
        fi;
      od;
    fi;
    #
    kk:=kk+1;
    kkprimnS:=kk^2*primnS;
    kkrSdualprim:=kkrSdualprim+rSdualprim;
    #
  od; #while kkprimnS>=-2 do
  #
  if passingLwalls=[] then beep(91911); fi;
  passingLwalls:=Set(passingLwalls);
  projLwalls:=passingLwalls*projS;
  Add(projLwalls, adjrec.enteringwall);
  if Rank(projLwalls)<>1 then beep(919229); fi;
  #
  perturbdegs:=List(passingLwalls, tr->tr/(nowweyldual*tr));
  SortParallel(perturbdegs, passingLwalls);
	# The ordering.
  if Set(passingLwalls, tr->tr*GramL*tr)<>[-2] then beep(61981); fi; #check
  if Set(passingLwalls, tr->tr*nowweyldual>0)<>[true] then beep(622981); fi; #check
  #adjrec.ADETypepassingLwalls:=Collected(GetRootType(GramL, passingLwalls));
  #
	reflmatL:=function(M, rv)
		# the product  M*(the reflection into (rv)\sperp)
   	local tv, rvdual;
  	rvdual:=rv*GramL;
  	return(List(M, tv->tv+(tv*rvdual)*rv));
  end;
  #
	tg:=IdentityMat(26);
	for rv in passingLwalls do #The order of passingLwalls mattres.
		  tg:=reflmatL(tg, rv);
	od;
  if TMTTmult(tg, GramL)<>GramL then beep(551852); fi;
  #
  #adjrec.opptg:=tg;
  newweyl:=nowweyl*tg;
  #if not IsWeylVect(GramL,newweyl) then beep(181822); fi; #check
  newweyldual:=newweyl*GramL;
  if Set(passingLwalls, tr->tr*newweyldual<0)<>[true] then beep(6442981); fi; #check
  if not IsEqualSet(passingLwalls*tg, (-1)*passingLwalls) then beep(6422981); fi; #check
  if newweyl*projS=nowweyl*projS then beep(91912); fi;
  adjrec.newweyl:=newweyl;
  #
  return(adjrec);
  #
end; #the end of MakeAdjRec.


MakeNonPerAutCham:=function(SXdata, chamrec) #NonPer means non-periodic
  local ii, nonperaut, beep, task,  GramS, nonperautrecs, tg;
  #
  beep:=function(beepnumb)
    localbeep("MakeNonPerAutCham", beepnumb); Error();
  end;
  #
  GramS:=SXdata.GramSX;
  nonperaut:=[];
  #
  task:=function(iis)
    local image, tg, ii;
    image:=List(iis, ii->chamrec.wallsS[ii]);
    tg:=chamrec.basisSinv*image;
    if IsIntMat(tg) then
      if TMTTmult(tg, GramS)<>GramS then beep(18182); fi; #for check
      if  IsEqualSet(chamrec.wallsS*tg, chamrec.wallsS) then
        Add(nonperaut, tg);
      fi;
    fi;
    return(true);
  end;
  #
  FindAllSubconfigstask(chamrec.targetM, chamrec.bigM, task);
  if not SXdata.theidS in nonperaut then beep(518511);fi;
  nonperautrecs:=[];
  for tg in nonperaut do
    Add(nonperautrecs, rec(g:=tg, qg:=OLtoOqL(tg, SXdata.discrecS)));
  od;
  return(nonperautrecs);
end; # the end of MakeNonPerAutCham


AreNonPerIsomChams:=function(SXdata, chamrec1, chamrec2)
  local ii, isoms, beep, task, GramS;
  # This is almost the same as MakeNonPerAutCham above.
  # This returns an isom tg such that
  # IsEqualSet(chamrec1.wallsS*tg, chamrec2.wallsS) if tg exists,
	# and it resurns false if not.
  #
  beep:=function(beepnumb)
    localbeep("AreNonPerIsomChams", beepnumb); Error();
  end;
  #
  if chamrec1.wallnrms<>chamrec2.wallnrms then return(false);fi;
  if chamrec1.fp<>chamrec2.fp then return(false); fi;
  #
  GramS:=SXdata.GramSX;
  #
  if chamrec1.weyl=chamrec2.weyl then return(SXdata.theidS); fi;
  #
  isoms:=[];
  #
  task:=function(iis)
    local image, tg, ii;
    image:=List(iis, ii->chamrec2.wallsS[ii]);
    tg:=chamrec1.basisSinv*image;
    if TMTTmult(tg, GramS)<>GramS then beep(18182); fi; #for checks
    if IsIntMat(tg) and IsEqualSet(chamrec1.wallsS*tg, chamrec2.wallsS) then
      Add(isoms, tg);
      return(true);
    fi;
    return(false);
  end;
  #
  FindASubconfigtask(chamrec1.targetM, chamrec2.bigM, task);
  if isoms=[] then return(false);
  else return(isoms[1]);
  fi;
end; #the end of AreNonPerIsomChams


IsNewCham:=function(SXdata, chamrecs, chamrec, resultrec)
  #
  local beep, GramS, tg, oldchamrec, old2pos, old2chamrec,ttg2,qtg2,
  isnew, qtg, tginv, qtginv, id, nonpertgrec, aut, oldchamautg, ttg, oldpos,
	autgrec;
  #
  # the output is
  # [is new by nonperiisom, is new by periisom]
	# (nonperisom is an isom of cahmbers that may not satisfy the period condition):
  # [false, false], [false, true], or [true, true].
  # At this point, chamrec is NOT in chamrecs.
  # resultrec is used to put tg in resultrec.generators
  #
  beep:=function(beepnumb)
    localbeep("IsNewCham", beepnumb); Error();
  end;
  #
  GramS:=SXdata.GramSX;
  chamrec.nonperisnew:=true; # initial set
  #
  oldpos:=0;
  for oldchamrec in chamrecs do
    oldpos:=oldpos+1;
    if  oldchamrec.pos<>oldpos then beep(473376); fi;
    if  oldchamrec.pos=chamrec.pos then beep(4764276); fi;
    if not oldchamrec.nonperisnew then continue; fi;
		# in or oldchamrec in chamrecs do
    #
    tg:=AreNonPerIsomChams(SXdata, oldchamrec, chamrec);
    # Since oldchamrec.bigM has been unbounded,
    # the order of  "oldchamrec, chamrec" should not be reversed.
		if tg=false then continue; fi; # in   for oldchamrec in chamrecs do
		# From now we have tg<>false.
		# chamrec is not new by nonperisom.
		# We do every task and return [false, false] or [false, true].
    chamrec.nonperisnew:=false;
    chamrec.nonperisomwith:=oldchamrec.pos;
    chamrec.nonperisomby:=tg; # from oldchamrec to chamrec
    qtg:=OLtoOqL(tg, SXdata.discrecS);
    tginv:=InverseMat(tg);  # from chamrec to oldchamrec
    qtginv:=OLtoOqL(tginv, SXdata.discrecS);
    chamrec.nonperisombyinv:=tginv;
    chamrec.nonperisombyq:=qtg; # from oldchamrec to chamrec
    chamrec.nonperisombyinvq:=qtginv; # from  chamrec to oldchamrec
		#
		# We decide which is the case: [false, false] or [false, true].
    for old2pos in oldchamrec.nonperisomwiths do
      old2chamrec:=chamrecs[old2pos];
      if not old2chamrec.isnew then beep(691692); fi;
			if old2chamrec.nonperisnew then
				if old2pos<>oldpos then beep(777692);fi;
			fi;
			if chamrec.nonperisomwith<>oldpos then beep(97654); fi;
      for nonpertgrec in oldchamrec.nonperaut do
        qtg2:=(old2chamrec.nonperisombyinvq*nonpertgrec.qg*qtg) mod 12;
        # old2cham to oldcham to nowcham.
				# Here we use the fact that discgSX is Z/12Z.
        if qtg2 in SXdata.periodpreservingqgs then
          chamrec.isnew:=false;
          chamrec.isomwith:=old2pos;
          ttg2:=old2chamrec.nonperisombyinv*nonpertgrec.g*tg;
          if not IsEqualSet(old2chamrec.wallsS*ttg2,chamrec.wallsS) then
            beep(696278);
          fi;
          if ttg2<>SXdata.theidS then
            Add(resultrec.generators, ttg2);
          fi;
          chamrec.isomby:=ttg2;
					chamrec.isnew:=false;
          chamrec.isnewflags:=[false, false];
          Unbind(chamrec.basisS);
          Unbind(chamrec.basisSinv);
          Unbind(chamrec.bigM);
          Unbind(chamrec.targetM);
          Unbind(chamrec.wallsSdual);
          # not new even by periisom.
					# Remark that being new by perisom is an easier condition than
					# being new by nonperisom.
          return([false, false]);
        fi;
      od; # for nonpertgrec in oldchamrec.nonperaut do
    od;# the end of for old2pos in oldchamrec.nonperisomwiths do
    #
    # Recall that we have tg<>false, and hence chamrec is not new in nonperiisom.
    # By the above, now we have: new by periisom
    #
		chamrec.isnew:=true;
    Add(oldchamrec.nonperisomwiths, chamrec.pos);
    aut:=[];
    for oldchamautg in oldchamrec.nonperaut do
      if (qtginv*oldchamautg.qg*qtg) mod 12 in SXdata.periodpreservingqgs then
        ttg:=tginv*oldchamautg.g*tg;
        Add(aut, ttg);
        if ttg<>SXdata.theidS then
          Add(resultrec.generators, ttg);
        fi;
      fi;
    od;
    chamrec.aut:=aut;
		chamrec.isnew:=true;
    chamrec.isnewflags:=[false, true];
    Unbind(chamrec.basisS);
    Unbind(chamrec.basisSinv);
    Unbind(chamrec.bigM);
    Unbind(chamrec.targetM);
    Unbind(chamrec.wallsSdual);
    return([false, true]);
		#
  od; # the end of for oldchamrec in chamrecs do
  #
  # now we have: new even by nonperiisom
  #
  chamrec.nonperisnew:=true;
	chamrec.isnew:=true;
  chamrec.nonperisomwith:=chamrec.pos;
  chamrec.nonperisomby:=SXdata.theidS;
  chamrec.nonperisombyinv:=SXdata.theidS;
  chamrec.nonperisombyq:=SXdata.theidqS;
  chamrec.nonperisombyinvq:=SXdata.theidqS;
  chamrec.nonperisomwiths:=[chamrec.pos];
  chamrec.nonperaut:=MakeNonPerAutCham(SXdata,chamrec);
  aut:=[];
  for autgrec in chamrec.nonperaut do
    if autgrec.qg in SXdata.periodpreservingqgs then
      Add(aut, autgrec.g);
      if autgrec.g<>SXdata.theidS then
        Add(resultrec.generators, autgrec.g);
      fi;
    fi;
  od;
  Unbind(chamrec.bigM);
  chamrec.aut:=aut;
  chamrec.isnewflags:=[true, true];
  return([true, true]);
end; #the end of IsNewCham


NewMakeChamRec:=function(SXdata, embSXLdata, weyl, chamrecs, frompos, enteringwall, resultrec)
	# We make a new chamrec from weyl.
  local chamrec, beep,GramS, nSaSvSss, nSaSvSs,  vSs,  GramSdual,
	rankS, linforms, vSlifts,walllfs,wallsS, wallsSdual, basisS, basisSinv,
	bigM, targetM, isnewflags, oldchamrec, lf, anisom, adjrecs, spanflag,
  adjrec, entflag,  walllf, fp, flatfps, xx, ffp, poss, pos, newposs, tgenvs,
	iniwallsS, iniwallsSdual, inifp, makingbasisS;
  #
  beep:=function(beepnumb)
    localbeep("NewMakeChamRec", beepnumb); Error();
  end;
  #
  nSaSvSss:=Make_nSaSvSss(embSXLdata,weyl);
  vSs:=[];
  for nSaSvSs in nSaSvSss do
    for vSlifts in nSaSvSs.vSliftss do
      Add(vSs, vSlifts[1]);
    od;
  od;
  #
  GramS:=SXdata.GramSX;
  GramSdual:=SXdata.GramSXdual;
  rankS:=Length(GramS);
  linforms:=Set(vSs*GramS, lf->MakePrimitiveIntVect(lf)[2]);
	#
  iniwallsSdual:=SelectWalls(linforms); # This is the most important part
	#
  iniwallsS:=iniwallsSdual*GramSdual;
  if enteringwall<>"ini" then
    if not (-1)*enteringwall in iniwallsS then beep(6291112); fi;
  fi;
  if Rank(iniwallsS)<>rankS then beep(91912); fi;
	#
	flatfps:=List(TMTTmult(iniwallsS, GramS), Collected);
	inifp:=Collected(flatfps);
	SortParallel(List(inifp, xx->xx[2]), inifp);
	poss:=[];
  spanflag:=false;
	makingbasisS:=0;
	for ffp in inifp do
		# This goes to the end even after spanflag is made true.
		# Hence poss becomes a rearranged [1..Length(iniwallsS)].
		# We make basisS and others ON THE WAY once.
		newposs:=Positions(flatfps, ffp[1]);
		if Length(newposs)<>ffp[2] then beep(5922); fi;
		Append(poss, newposs);
    if not spanflag then
      tgenvs:=List(poss, pos->iniwallsS[pos]);
      if Rank(tgenvs)=rankS then
        basisS:=FindSubvsWithRank(tgenvs, rankS)[1];
        basisSinv:=InverseMat(basisS);
        bigM:=TMTTmult(tgenvs, GramS);
        targetM:=TMTTmult(basisS, GramS);
        spanflag:=true;
				makingbasisS:=makingbasisS+1;
      fi;
    fi;
	od; #the end of for ffp in inifp do
	#
  if not spanflag then beep(661992); fi;
	if makingbasisS<>1 then beep(881992); fi;
	if Length(poss)<>Length(iniwallsS) then beep(9191); fi;
	if Set(poss)<>[1..Length(iniwallsS)] then beep(91991); fi;
	wallsS:=List(poss, pos->iniwallsS[pos]);
  wallsSdual:=List(poss, pos->iniwallsSdual[pos]);
	flatfps:=List(TMTTmult(wallsS, GramS), Collected);
  fp:=Collected(flatfps);
  #
  #
  chamrec:=rec(
    pos:=Length(chamrecs)+1,
    frompos:=frompos,
    enteringwall:=enteringwall,
    weyl:=weyl,
    nSaSvSss:=nSaSvSss,
    #nSaSnopss:=nSaSnopss,
    wallnrms:=Collected(List(wallsS, tv->tv*GramS*tv)),
    wallsS:=wallsS,
    wallsSdual:=wallsSdual,
    basisS:=basisS,
    basisSinv:=basisSinv,
    bigM:=bigM,
    targetM:=targetM,
		fp:=fp
  );
  #
  isnewflags:=IsNewCham(SXdata, chamrecs, chamrec, resultrec);
  #
  if isnewflags[2] then # chamrec is new by periisom
    adjrecs:=[];
    entflag:=0;
		#Printn("__________start making adjrecs  in MakeChamRec");
    for walllf in wallsSdual do
      if enteringwall<>"ini" and walllf=(-1)*enteringwall*GramS  then
        adjrec:=rec(enteringwall:=(-1)*enteringwall, israt:=false, backto:=frompos, isback:=true);
        entflag:=entflag+1;
      else
        adjrec:=MakeAdjRec(embSXLdata, weyl, walllf);
				adjrec.isback:=false;
      fi;
      Add(adjrecs, adjrec);
    od;
		#Printn("__________done making adjrecs  in MakeChamRec");
    if entflag=0 then if enteringwall<>"ini" then beep(55763); fi;
    else if entflag<>1 then beep(52828282); fi;
    fi;
    chamrec.adjrecs:=adjrecs;
    Unbind(chamrec.bigM);
  fi; # the end of if isnewflags[2] then
  #
  return(chamrec);
end; #the end of NewMakeChamRec


BorcherdsL26AF:=function(SXdata, embSXLdata, tempname)
  local beep, recnames, GramL26, LGrec, embSXL26, GramSX, rhoX, embgeneralampleSX,
	weyl0,  nRvRsrec, nSaSvSss, chamrecs, vSs, nSaSvSs, nS, aS, vSlifts, lf, xx,
	linforms,  inichamrec, autnops, champos,isnewnops,donesnops, nowcham, ii,
	adjrec, newchamrec, tchamrec, gens, ccc, backnum, nonperisnewnops, resultrec;
  #
  beep:=function(beepnumb)
    localbeep("BorcherdsL26AF", beepnumb); Error();
  end;
  #
  GramL26:=embSXLdata.GramL26;
  LGrec:=MakeLGrecFromEmb1(GramL26, embSXLdata.embSXL26);
  embSXLdata.LGrec:=LGrec;
  embSXL26:=LGrec.emb1;
  if embSXLdata.embSXL26<>embSXL26 then beep(6786278); fi;
  GramSX:=LGrec.Gram1;
  if SXdata.GramSX<>GramSX then beep(67981682); fi;
  SXdata.GramSXdual:=InverseMat(GramSX);
  rhoX:=Length(GramSX);
  SXdata.theidS:=IdentityMat(rhoX);
  SXdata.theidqS:=OLtoOqL(SXdata.theidS, SXdata.discrecS);
	if SXdata.theidqS<>[[1]] then beep(7253); fi;
  embgeneralampleSX:=embSXLdata.embgeneralampleSX;
  #
  # ampleSX is not on a wall of NX, but may be on a wall of an L26/SX chamber.
  # embgeneralampleSX is not on a wall of any L26/SX chamber.
  # embgeneralampleSX is more general than ampleSX.
  #
  embSXLdata.ampleSXdual:=embgeneralampleSX*GramSX;
  weyl0:=embSXLdata.weyl0;
  embSXLdata.nRvRsrec:=Make_nRvRsRec(embSXLdata);
  nSaSvSss:=Make_nSaSvSss(embSXLdata, weyl0);

  #
  resultrec:=rec();
  resultrec.generators:=[];
  Printn("start making chamrecs");
  inichamrec:=NewMakeChamRec(SXdata, embSXLdata, weyl0, [], 0, "ini", resultrec);
  #
  # check the initial interior point (embgeneralampleSX)
  #
  vSs:=Union(List(nSaSvSss, nSaSvSs->List(nSaSvSs.vSliftss, vSlifts->vSlifts[1])));
  linforms:=Set(inichamrec.wallsS*GramSX);
  if linforms<>Set(linforms, lf->GetPrimitiveIntVect(lf)) then
    beep(69791);
  fi;
  if Set(linforms*embgeneralampleSX, xx->xx>0)<>[true] then beep(815218); fi;
  #
	########### the end of heck the initial interior point
	#
  Unbind(inichamrec.nSaSvSss);
	Unbind(inichamrec.wallsSdual);
  if Length(inichamrec.aut)=1 then
    if inichamrec.aut[1]<>SXdata.theidS then beep(62269693); fi;
    inichamrec.aut:="trivial"; # to make the data small
    autnops:=1;
  else autnops:=Length(inichamrec.aut);
  fi;
  chamrecs:=[inichamrec];
  resultrec.chamrecs:=chamrecs;
  #
	################## Here we start the main part.
  champos:=1;  nonperisnewnops:=1; isnewnops:=1; donesnops:=0;
  while champos<=Length(chamrecs) do # This is the main part
    nowcham:=chamrecs[champos];
    if nowcham.isnew then
      ccc:=0;
      backnum:=0;
      for adjrec in nowcham.adjrecs do
        ccc:=ccc+1;
        Printn("___adjrec in nowcham.adjrecs", ccc, "in", Length(nowcham.adjrecs), ":champos", champos, "in", Length(chamrecs));
        if (not adjrec.israt) and (not adjrec.isback) then
          newchamrec:=NewMakeChamRec(SXdata, embSXLdata, adjrec.newweyl,
					chamrecs, nowcham.pos, adjrec.enteringwall, resultrec);
          if not (-1)*adjrec.enteringwall in newchamrec.wallsS then beep(811233); fi;
          if newchamrec.isnew then
            if Length(newchamrec.aut)=1 then
              if newchamrec.aut[1]<>SXdata.theidS then beep(69869693); fi;
              newchamrec.aut:="trivial"; # to make the data small
              autnops:=1;
            else
							autnops:=Length(newchamrec.aut);
            fi;
          fi;
          #
          if newchamrec.isnewflags=[ false,  false ] then
            Unbind(newchamrec.wallnrms);
						Unbind(newchamrec.nSaSvSss);
						Unbind(newchamrec.nonperisombyinv);
            Unbind(newchamrec.nonperisombyinvq);
            Unbind(newchamrec.nonperisombyq);
            # Unbind(newchamrec.wallsS);
            #"enteringwall", "fp", "frompos", "isnew", "isnewflags", "isomby", "isomwith", "nSaSvSss",
            # "nonperisnew", "nonperisomby", "nonperisombyinv", "nonperisombyinvq",
            #              "nonperisombyq", "nonperisomwith", "pos", "wallnrms", "wallsS", "weyl"
          #
          elif newchamrec.isnewflags=[ false, true ] then
            Unbind(newchamrec.nSaSvSss);
          elif newchamrec.isnewflags=[true, true] then
            Unbind(newchamrec.nSaSvSss);
            Unbind(newchamrec.wallsSdual);

          fi;

          Add(chamrecs, newchamrec);

              Printn("___isnewflags", newchamrec.isnewflags, "aut", autnops);
            fi;
          else
            Printn("___isnewflags", newchamrec.isnewflags);
          fi;
        else #if (not adjrec.israt) and (not adjrec.isback) then
          if adjrec.israt then Printn("___rat"); fi;
          if adjrec.isback then Printn("___back"); backnum:=backnum+1;fi;
        fi;
      od; #the ned of for adjrec in nowcham.adjrecs do
      if backnum<>1 then
			 	if champos<>1 then beep(919922); fi;
			fi;
      #
      donesnops:=donesnops+1;
      resultrec.generators:=Set(resultrec.generators);
      Printn("_", champos, Length(chamrecs),"nonperisnews", nonperisnewnops,
         "isnews", isnewnops, "dones", donesnops, "rest", isnewnops-donesnops);
      #savedataas(resultrec, tempname); #resultrec.chamrecs:=chamrecs has been done above
    fi;
    champos:=champos+1;
  od; #the end of while champos<=Length(chamrecs) do
	#
  Printn("#########");
  Printn("making chamrecs: done");
