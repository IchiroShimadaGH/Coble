SmallADEDiagramAutOrder:=function(chunk)
	local ADEtype, smallaut, type, rank, succ, order;
  #
	ADEtype:=chunk[1];
	type:=ADEHrec.(ADEtype).type;
	rank:=ADEHrec.(ADEtype).rank;
	succ:=chunk[2];
	if type='A' then
		if rank=1 then
			smallaut:=1;
		else
			smallaut:=2;
		fi;
	elif type='D' then
		if rank=4 then
			smallaut:=6;
		else
			smallaut:=2;
		fi;
	elif type='E' then
		if rank=6 then
			smallaut:=2;
		elif rank=7 then
			smallaut:=1;
    elif rank=8 then
      smallaut:=1;
		else beep(100);
		fi;
	else
		beep(200);
	fi;
	order:=smallaut^succ*Factorial(succ);
	return(order);
end;


ADEDiagramAutOrder:=function(typelist)
	#
	local typechunks, order;
	#
	typechunks:=ToMultiSet(typelist);
	order:=Product(List(typechunks, SmallADEDiagramAutOrder));
	return(order);
end;


SmallADEDiagramAutGens:=function(chunk)
	#
	#  chunk is [A_i or D_i or E_i, number]
	#
 	local beep, ADEtype, type, rank, succ, typelist, gens,
 	Idr, Zeror, totalrank, Idrest, tg1, tg2, poss, makemat, perm,
 	Gram, tg, ii;
 	#
 	beep:=function(numb)
 		Printn("wrong in  diagramautgenschunk", numb); Error();
 	end;
 	#
 	ADEtype:=chunk[1];
	type:=ADEHrec.(ADEtype).type;
	rank:=ADEHrec.(ADEtype).rank;
	succ:=chunk[2];
	typelist:=List([1..succ], ii->ADEtype);
	gens:=[];
	Idr:=IdentityMat(rank);
	Zeror:=NullMat(rank,rank);
	totalrank:=rank*succ;
	if succ>1 then
		Idrest:=IdentityMat(totalrank-rank);
	fi;
	#
	if type='A' then
		if rank=1 then
			#do nothing
		else
			tg1:=Reversed(Idr);
			if succ=1 then
			else
				tg1:=DiagonalMats([tg1, Idrest]);
			fi;
			Add(gens, tg1);
		fi;
	elif type='D' then
		poss:=Concatenation([2,1], [3..rank]);
		tg1:=List(poss, nn->MakeVectei(rank,nn));
		if rank=4 then
			poss:=[4,2,3,1];
			tg2:=List(poss, nn->MakeVectei(rank,nn));
			if succ=1 then
			else
				tg1:=DiagonalMats([tg1, Idrest]);
				tg2:=DiagonalMats([tg2, Idrest]);
			fi;
			Append(gens, [tg1, tg2]);
		else
			if succ=1 then
			else
				tg1:=DiagonalMats([tg1, Idrest]);
			fi;
			Add(gens, tg1);
		fi;
	elif type='E' then
		if rank=6 then
			poss:=[1,6,5,4,3,2];
			tg1:=List(poss, nn->MakeVectei(rank,nn));
			if succ=1 then
			else
				tg1:=DiagonalMats([tg1, Idrest]);
			fi;
			Add(gens, tg1);
		elif rank=7 then
			#do nothing
		elif rank=8 then
			#do nothing
		else
			beep(300);
		fi;
	else
		beep(200);
	fi;
	#
	if succ>1 then
		#
		makemat:=function(perm)
			local Idsucc, M, m, vL, a, vM, v;
			Idsucc:=IdentityMat(succ);
			m:=List([1..succ], ii->Idsucc[ii^perm]);
			M:=[];
			for v in m do
				vL:=[];
				for a in v do
					if a=0 then Add(vL, Zeror);
					elif a=1 then Add(vL, Idr);
					else beep(79);
					fi;
				od;
				vM:=ConcatMatss(vL);
				if M=[] then M:=vM;
				else
					Append(M, vM);
				fi;
			od;
			return(M);
		end;
		#
		tg1:=makemat((1,2));
		Add(gens, tg1);
		if succ>2 then
			perm:=PermList(CopyAdd([2..succ], 1));
			tg1:=makemat(perm);
			Add(gens, tg1);
		fi;
	fi;
	#
	Gram:=ADEHGram(typelist);
	for tg in gens do
		if TMTTmult(tg, Gram)<>Gram then beep(100); fi;
	od;
	if gens=[] then
		Add(gens, IdentityMat(ADEHrank(typelist)));
	fi;
	if SmallADEDiagramAutOrder(chunk)<>Size(Group(gens)) then beep(2000); fi;
	return(gens);
 end;

ADEDiagramAutGens:=function(typelist)
	#
  # typelist must be compatible with its chunk form.
  #
  #
	local beep, Gram, gens, chunks, tg, ranks, Ids,
	pos, ii, factors, rr,chunk, size, tgg, xx, ttlist,
  rts, rt, kk, nops, typelist2;
 	#
 	beep:=function(beepnumb)
 		localbeep("DiagramAutGens", beepnumb); Error();
 	end;
 	#
  Gram:=ADEHGram(typelist);
  #
  chunks:=[];
  rts:=[];
  typelist2:=[];
  for rt in typelist do
    if not rt in rts then
      nops:=OccurNumb(typelist, rt);
      Add(chunks, [rt, nops]);
      Add(rts, rt);
      for kk in [1..nops] do
        Add(typelist2, rt);
      od;
    fi;
  od;
  if typelist2<>typelist then beep(522112); fi;
  #
 	gens:=[];
 	ranks:=List(chunks, chunk->ADEHrec.(chunk[1]).rank*chunk[2]);
 	Ids:=List(ranks, rr->IdentityMat(rr));
 	#
 	pos:=0;
 	for chunk in chunks do
 		pos:=pos+1;
 		factors:=[];
 		for ii in [1..Length(chunks)] do
 			if ii<>pos then
 				Add(factors, [Ids[ii]]);
 			else
 				Add(factors, SmallADEDiagramAutGens(chunks[pos]));
 			fi;
 		od;
 		for tgg in Cartesian(factors) do
 			tg:=DiagonalMats(tgg);
 			Add(gens, tg);
 		od;
 	od;
 	#
 	for tg in gens do
		if TMTTmult(tg, Gram)<>Gram then beep(100); fi;
	od;
	if gens=[] then
		Add(gens, IdentityMat(ADEHrank(typelist)));
	fi;
	size:=Size(Group(gens));
	if Product(List(chunks, chunk->SmallADEDiagramAutOrder(chunk)))<>size then
		beep(2000);
	fi;
	if ADEDiagramAutOrder(typelist)<>size then
		beep(2001);
	fi;
	return(gens);
 end;
