RandomFindvs:=function(Gram, p)
	#
	local n, tv, aa, bb, tv2, sol;
	#
	n:=Length(Gram);
	while true do
		tv:=RandomVect(n) mod p;
		aa:=tv*Gram*tv;
		if aa mod p =0 then
			bb:=-aa/p/2 mod p;
			tv2:=SolAbModp([tv*Gram], [bb], p).asol;
			sol:=tv+p*tv2;
			if sol*Gram*sol mod p^2<>0 then beep(44233); fi;
			if not IsZeroVect(sol mod p) then
				return(sol);
			fi;
		fi;
	od;
end;


Kneighbor:=function(Gram, v, p, det)
	local n, zv, Lv, newbasis, newGram;
	#
	n:=Length(Gram);
	Lv:=SolAbModp([v*Gram], [0], p).nullvects;
	newbasis:=p^2*IdentityMat(n);
	Append(newbasis, p*Lv);
	Add(newbasis, v);
	newbasis:=HermiteNormalFormIntegerMat(newbasis);
	newbasis:=RemoveZeroVectsIntMat(newbasis);
	newGram:=(1/p^2)*TMTTmult(newbasis, Gram);
	if not IsEvenLattice(newGram) then beep(53443); fi;
	if DeterminantIntMat(newGram)<>det then beep(43332); fi;
	return(newGram);
end;

DoesContainARoot:=function(posGram)
  return(ShortestVectors(posGram, 2).vectors<>[]);
end;



KneserP:=function(Gram, iniseed)
	#
	local beep, n, det, p, tp, negcps, rest, mass,seed,
	localdata, flag, dd, c, types, isnew,  tv, oldGram, newGram,
	kneserrec, ogrec, ccc, isnewf, classrec, isomclasses, rootflag;
	#
	beep:=function(beepnumb)
		localbeep("Kneser", beepnumb); Error();
	end;
	#
	Reset(GlobalMersenneTwister, iniseed);
	seed:=iniseed;
	#
	n:=Length(Gram);
	if SignatureQ(Gram)<>[n,n,0] then beep(6345); fi;
	det:=DeterminantIntMat(Gram);
	localdata:=LocalDataRatQF(Gram);
	dd:=localdata.d0;
	negcps:=localdata.negativecps;
	tp:=3;
	flag:=false;
	while not flag do
		if IsPrime(tp) then
			if det mod tp<>0 then
				if tp in negcps then c:=-1; else c:=1; fi;
				if IsZeroRepresentableByFp(tp, n, dd, c) then
						p:=tp;
						flag:=true;
				fi;
			fi;
		fi;
		tp:=tp+1;
	od;
	#
	#
	mass:=Mass(Gram);
	Printn("mass", mass);
	#
	types:=[MakePosLatType(Gram)];
	ogrec:=OGPosLat(Gram, 1,1);
	rest:=mass-1/ogrec.order;
	rootflag:=DoesContainARoot(Gram);
	Printn("initial", ogrec.order, "rest", rest, rootflag);
	classrec:=rec(Gram:=Gram, og:=ogrec.order, rootflag:=rootflag);
	isomclasses:=[classrec];
	#
	isnew:=function(newGram)
		local newtype, oldtype, tg, rootflag;
		newtype:=MakePosLatType(newGram);
		for oldtype in types do
			tg:=AreIsomPosLats(oldtype, newtype, seed,  1, 1000); #1000 is the tab
			if tg<>false then return(false); fi;
		od;
		if Mass(newGram)<>mass then beep(443445); fi;
		ogrec:=OGPosLat(newGram,seed,1);
		rootflag:=DoesContainARoot(newGram);
		classrec:=rec(Gram:=newGram, og:=ogrec.order, rootflag:=rootflag);
		Add(isomclasses, classrec);
		Add(types, newtype);
		return(classrec);
	end;
	#
	ccc:=0;
	while rest>0 do
		ccc:=ccc+1;
		oldGram:=Random(isomclasses).Gram;
		tv:=RandomFindvs(oldGram, p);
		newGram:=Kneighbor(oldGram, tv, p, det);
		isnewf:=isnew(newGram);
		if isnewf<>false then
			rest:=rest-1/isnewf.og;
			Printn("new", ccc, "order", isnewf.og, "rest", rest,  isnewf.rootflag);
			#
		else
			if ccc mod 1000=0 then
				Printn("not new", ccc);
			fi;
		fi;
		seed:=seed+1;
		Reset(GlobalMersenneTwister, seed);
	od;
	if rest<>0 then beep(326523653); fi;
	Printn("___ Kneser done", mass);
	kneserrec:=rec(
		mass:=mass,
		isomclasses:=isomclasses
	);
	return(kneserrec);
end;
