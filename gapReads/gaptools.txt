IsSquareRootRat:=function(a)
	local da, na;
	na:=NumeratorRat(a);
	if na<0 then return(false); fi;
	da:=DenominatorRat(a);
	if RootInt(da)^2<>da then return(false); fi;
	if RootInt(na)^2<>na then return(false); fi;
	return(RootInt(na)/RootInt(da));
end;

Ordp:=function(n, p)
	#
	# The largest integer e such that p^e divides the integer n.
	#
	local e, nn;
	#
	if n=0 then Printn("wrong_in_Ordp", n); Error(); fi;
	#
	e:=0;
	nn:=n;
	while nn mod p=0 do
		e:=e+1;
		nn:=nn/p;
	od;
	return(e);
end;

ClassifyByTags:=function(list, tagfunc)
	#
	# tagfunc should not destroy the input,
	# otherwise  list will be changed when tags:=List(list, tagfunc).
	#
	# This classifies the elements of list according to the value of tagfunc.
	# The output is the list of [tagvalue, the list of elms x with tagfunc(x)=tagvalue].
	# This list is sorted according to tag-value.
	#
	#
	local tags, dftags, tlist, tag, ttlist, ii;
	tags:=List(list, tagfunc);
	dftags:=DuplicateFreeList(tags);
	Sort(dftags);
	tlist:=[];
	for tag in dftags do
		ttlist:=[];
		for ii in [1..Length(list)] do
			if tags[ii]=tag then
				AddCopy(ttlist, list[ii]);
			fi;
		od;
		Add(tlist, [tag, ttlist]);
	od;
	return(tlist);
end;

OrbitdecompByConnects:=function(n, connects, prevmerge)
	#
	#
	local orbs, ii, newconnects, connii, tconn, jj, beep,
	neworb, neworbs, oldorb, donekks, newconnects2, tt;
	#
	beep:=function(numb)
		Printn("wrong OrbitdecompByConnects", numb); Error();
	end;
	#
	newconnects:=List(connects);
	#
	for tt in [1..prevmerge] do
		newconnects2:=[];
		for ii in [1..n] do
			connii:=[ii];
			for tconn in newconnects do
				if ii in tconn then
					for jj in tconn do
						if jj<>ii then Add(connii, jj);fi;
					od;
				fi;
			od;
			connii:=Set(connii);
			Add(newconnects2, connii);
		od;
		newconnects:=Set(newconnects2);
	od;
	#
	orbs:=[];
	donekks:=[];
	#
	for connii in newconnects do
		neworbs:=[];
		if Intersection(donekks, connii)=[] then
			Add(orbs, connii);
		else
			neworbs:=[];
			neworb:=List(connii);
			for oldorb in orbs do
				if Intersection(oldorb, connii)=[] then
					Add(neworbs, oldorb);
				else
					Append(neworb, oldorb);
				fi;
			od;
			neworb:=Set(neworb);
			Add(neworbs, neworb);
			orbs:=neworbs;
		fi;
		Append(donekks, connii);
		donekks:=Set(donekks);
	od;
	#
	if not IsDuplicateFree(Flat(orbs)) then beep(100); fi;
	if Length(Set(Flat(orbs)))<>n then beep(200); fi;
	if not IsEqualSet(Concatenation(orbs), [1..n]) then beep(300); fi;
	#
	return(orbs);
end;

lexsorter:=function(v1, v2)
	#
	# if v1 is less than v2 by lex, then -1
	# if v1 is equal to v2 , then 0,
	# if v1 is larger than v2 then 1.
	#
	local ii, c1, c2, comp;
	if IsList(v1) then
		for ii in [1..Length(v1)] do
			c1:=v1[ii]; c2:=v2[ii];
			comp:=lexsorter(c1, c2);
			if comp<>0 then return(comp); fi;
		od;
		return(0);
	else
		if v1<v2 then return(-1);
		elif v1>v2 then return(1);
		else return(0);
		fi;
	fi;
	BeepStr("lexsorter 10");
end;

GroupGeneratedBy:=function(gens, productfunction, identityelement)
	#
	local G, job;
	G:=[List(identityelement)];
	#
	# induction by length of the words.
	#
	job:=function(Gleng)
		local newelms, oldelm, tgen, tg;
		newelms:=[];
		for oldelm in Gleng do
			for tgen in gens do
				tg:=productfunction(oldelm , tgen);
				if not tg in G then
					Add(newelms, tg);
				fi;
			od;
		od;
		if newelms<>[] then
			newelms:=Set(newelms);
			Append(G, newelms);
			job(newelms);
		fi;
	end;
	job([List(identityelement)]);
	G:=Set(G);
	return(G);
end;

SqRoot:=function(xx)
	local fxx, sqnxx, sqdxx, sqxx, dxx, nxx;
	if  not (IsFloat(xx)) and xx<0 then
		Printn("wrong in SqRoot"); Error();
	fi;
	if IsInt(xx) then
		sqxx:=RootInt(xx);
		if sqxx^2=xx then
			return(sqxx);
		fi;
	fi;
	if IsRat(xx) then
		dxx:=DenominatorRat(xx);
		nxx:=NumeratorRat(xx);
		sqdxx:=RootInt(dxx);
		sqnxx:=RootInt(nxx);
		if sqdxx^2=dxx and sqnxx^2=nxx then
			return(sqnxx/sqdxx);
		fi;
	fi;
	fxx:=Float(xx);
	return(Exp(Log(fxx)/2));
end;

SqIntRange:=function(vv, uu)
	#
	# [min..max] is the range of x in Z such that (x-vv)^2<=uu
	#
	local beep,  max, xx, min;
	#
	beep:=function(numb)
		Printn("wrong SqIntRange", numb); Error();
	end;
	#
	if uu<0 then beep(587557); fi;
	xx:=FloorRat(vv);
	while true do
		xx:=xx+1;
		if xx>vv and (xx-vv)^2>uu then
			max:=xx-1;
			break; #from while true do
		fi;
	od;
	xx:=FloorRat(vv)+1;
	while true do
		xx:=xx-1;
		if xx<vv and (xx-vv)^2>uu then
			min:=xx+1;
			break; #from while true do
		fi;
	od;
	return([min, max]);
end;

EquivClasses:=function(L, isoms)
  #
  # decomposition of the set Copyright (c)
  # into the equivalence classes
  # by the equivalence relation generated by
  # a ~ b  iff [a, b] in isoms
  #
  local beep, tisoms, equivs, aas, aa, flag, bbs;
  #
  beep:=function(beepnumb)
    localbeep("EquivClasses", beepnumb); Error();
  end;
  tisoms:=List(isoms);
  equivs:=List(SetMinus(L, Union(isoms)), aa->[aa]);
  while tisoms<>[] do
    aas:=Remove(tisoms);
    flag:=true;
    for bbs in tisoms do
      if Intersection(aas, bbs)<>[] then
        Append(bbs, aas);
        flag:=false;
        break;
      fi;
    od;
    if flag then
      Add(equivs, Set(aas));
    fi;
  od;
  #
  if not Sum(equivs, Length)=Length(Set(L)) then beep(391681); fi;
  if not IsEqualSet(Union(equivs), L) then beep(265181); fi;
  return(equivs);
end;

		localbeep("NewClassifyByTags", beepnumb); Error();
	end;
	#
  tags:=List(list, tagfunc);
	tagsset:=Set(tags);

ConnectedCompsGraph:=function(adjacencymat)
  #
  local beep, size, conncomps, dones, nn, newconncomp,
  kk, mm, pp;
  #
  beep:=function(beepnumb)
    localbeep("ConnectedCompsGraph", beepnumb); Error();
  end;
  #
  if adjacencymat<>TransposedMat(adjacencymat) then beep(726353); fi;
  #
  size:=Length(adjacencymat);
  conncomps:=[];
  dones:=[];
  #
  for nn in [1..size] do
    if not nn in dones then
      newconncomp:=[nn];
      kk:=1;
      while kk<=Length(newconncomp) do
        mm:=newconncomp[kk];
        for pp in Positions(adjacencymat[mm], 1) do # We cannot use Union!
          if not pp in newconncomp then
            Add(newconncomp, pp);
          fi;
        od;
        kk:=kk+1;
      od; #while kk<=Length(newconncomp) do
      #
      newconncomp:=Set(newconncomp);
      Add(conncomps, newconncomp);
      if Intersection(dones, newconncomp)<>[] then beep(3321929); fi;
      Append(dones, newconncomp);
      #
    fi;
  od; #for nn in [1..size] do
  #
  Sort(dones);
  if dones<>[1..size] then beep(445662); fi;
  return(conncomps);
end;

AbsRat:=function(rat)
  return(AbsInt(NumeratorRat(rat))/AbsInt(DenominatorRat(rat)));
end;

SignRat:=function(rat)
  local nrat, drat;
  nrat:=NumeratorRat(rat);
  drat:=DenominatorRat(rat);
  if not drat>0 then beep(11638587); fi;
  if nrat>0 then return(1);
  elif nrat=0 then return(0);
  elif nrat<0 then return(-1);
  else beep(525278);
  fi;
end;

OrbitdDecompByPerms:=function(nn, perms)
  #
  # perms is a list of permutations of [1..nn].
  # orbs is the list of orbits of the action of Group(permutations) on [1..nn].
  #
  local orbs, dones, tg, ii, jj, cc, torb, jjtg, forbs, beep;
  #
  beep:=function(beepnumb)
    localbeep("OrbitdDecompByPerms", beepnumb); Error();
  end;
  #
  orbs:=[];
  dones:=[];
  #
  for ii in [1..nn]  do 
    if ii in dones then continue; fi;
    torb:=[ii];
    cc:=0;
    for jj in torb do 
      cc:=cc+1;
      for tg in perms do 
        jjtg:=jj^tg;
        if not jjtg in torb then Add(torb, jjtg); fi;
      od;
    od;
    if cc<>Length(torb) then beep(91919); fi;
    Add(orbs, torb);
    Append(dones, torb);
  od;
  #
  Sort(dones);
  if dones<>[1..nn] then beep(71967981); fi;
  orbs:=List(orbs, Set);
  forbs:=Flat(orbs);
  Sort(forbs);
  if forbs<>[1..nn] then beep(71955981); fi;
  #
  return(orbs);
  #
end;
