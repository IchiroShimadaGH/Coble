Make_OrthogonalRoots:=function(GramS, pol)
	return(AffESstd(GramS, pol, 0, -2, true));
end;


FindRoottype:=function(rank, disc, rootnumb)
	local rttypes;
	rttypes:=[];
	if rootnumb=rank*(rank+1) then
		if disc=rank+1 then
			Add(rttypes, Concatenation("A", String(rank)));
		fi;
	fi;
	if rootnumb=2*rank*(rank-1) then
		if rank>=4 and disc=4 then
			Add(rttypes, Concatenation("D", String(rank)));
		fi;
	fi;
	if rootnumb=72 then
		if rank=6 and disc=3 then
			Add(rttypes, "E6");
		fi;
	fi;
	if rootnumb=126 then
		if rank=7 and disc=2 then
			Add(rttypes, "E7");
		fi;
	fi;
	if rootnumb=240 then
		if rank=8 and disc=1 then
			Add(rttypes, "E8");
		fi;
	fi;
	if rttypes=[] then
		Printn("no such root type", rank, disc, rootnumb); Error();
	elif Length(rttypes)>1 then
		Printn("cannot determine root type", rank, disc, rootnumb, rttypes); Error();
	fi;
	return(rttypes[1]);
end;



Make_OrthogonalRootSystem:=function(GramS, pol, perturbvects_or_ample)
	#
	# the result is the list of
	# [root_type,
	# list of roots sorted accroding to the standard Dynkin (in ADEHrec)].
	#
	local posRspol, Rspol, isposdeg, perturbvectsdual,  degsorter, orbs,
	kk, jj, mm, tr, tr2, tr3, trdeg, tr2deg, tr3deg, nz, orb, connjob,
	indecomproots, rttype, tGram, ADEGram, isom, data, r, rdual, decompflag,
	dones, originalposRspol, rtnumb;
	#
	Rspol:=Make_OrthogonalRoots(GramS, pol);
	if Rspol=[] then return([]); fi;
	if perturbvects_or_ample=[] then Printn("we need perturbvects_or_ample"); Error(); fi;
	#
	if IsList(perturbvects_or_ample[1]) then
		perturbvectsdual:=perturbvects_or_ample*GramS;
	else
		perturbvectsdual:=[perturbvects_or_ample]*GramS;
	fi;
	#
	#
	isposdeg:=function(v)
		local pvdual, deg;
		for pvdual in perturbvectsdual do
			deg:=pvdual*v;
			if deg>0 then return(true);
			elif deg<0 then return(false);
			fi;
		od;
		Printn("wrong perturbvects"); Error();
		return(true);
	end;
	#
	#posRspol:=Filtered(Make_OrthogonalRoots(GramS, pol), isposdeg); #corrected 2017 May 16
	posRspol:=Filtered(Rspol, isposdeg);
	#Printn(TMTTmult(posRspol, GramS));
	# this will be changed
	#
	#
	originalposRspol:=CopyMat(posRspol);
	dones:=[];
	#
	rtnumb:=0;
	data:=[];
	#
	while Length(posRspol)>0 do
		r:=Remove(posRspol);
		orb:=[r];
		connjob:=function(tr, trdual)
			local ttr2, intnumb;
			for ttr2 in CopyMat(posRspol) do
				if ttr2 in posRspol then
					intnumb:=ttr2*trdual;
					if AbsInt(intnumb)=1 then
						Add(orb, Remove(posRspol, Position(posRspol, ttr2)));
						#ttr2dual:=ttr2*GramS;
						connjob(ttr2, ttr2*GramS);
					elif intnumb<>0 then Printn("wrong intnumb in connjob"); Error();
					fi;
				fi;
			od;
		end;
		#
		connjob(r, r*GramS);
		#
		#Printn(Length(orb));
		#
		degsorter:=function(v1, v2)
			local diffv, pvdual, deg;
			diffv:=v1-v2;
			for pvdual in perturbvectsdual do
				deg:=pvdual*diffv;
				if deg<0 then return(true);
				elif deg>0 then return(false);
				fi;
			od;
			return(true);
		end;
		#
		Append(dones, orb);
		Sort(orb, degsorter);
		#
		indecomproots:=[];
		for kk in [1..Length(orb)] do
			tr:=orb[kk];
			trdeg:=perturbvectsdual*tr;
			decompflag:=false;
			#
			for jj in [1..kk-1] do
				tr2:=orb[jj];
				tr2deg:=perturbvectsdual*tr2;
				for mm in [1..jj] do
					tr3:=orb[mm];
					tr3deg:=perturbvectsdual*tr3;
					nz:=IniValnzErrorFree(tr2deg+tr3deg-trdeg);
					if nz>0 then break; fi; #for mm in
					if nz=0 then
						if tr2+tr3=tr then decompflag:=true; break; fi; #for mm in
					fi;
				od;
				if decompflag then break; fi; #for jj in
			od;
			if not decompflag then Add(indecomproots, tr); fi;
		od;
		#
		tGram:=TMTTmult(indecomproots, GramS);
		rttype:=FindRoottype(Length(indecomproots), AbsInt(DeterminantIntMat(tGram)), 2*Length(orb));
		ADEGram:=(-1)*ADEHrec.(rttype).Gram;
		isom:=FindASubconfig(tGram, ADEGram); #function(targetM, bigM)
		if isom=false or isom=fail then
			Printn("wrong in Make_PosOrthogonalRoots"); Error();
		fi;
		SortParallel(isom, indecomproots);
		#
		# SortParallel( list1, list2)
		# sorts  list1 in increasing order just as Sort  does.
		# In parallel it applies the same exchanges that are necessary
		# to sort list1 to list2.
		#
		# added on 5 May 2016.
		if TMTTmult(indecomproots, GramS)<>ADEGram then
			Printn("wrong in Make_PosOrthogonalRoots",  7); Error();
		fi;
		#
		rtnumb:=rtnumb+ADEHrec.(rttype).rootnumb;
		#
		Add(data, [rttype, indecomproots]);
		#
	od;
	#
	# added on 5 May 2016.
	Sort(data, function(xx, yy) return(indecompADEsorter(xx[1], yy[1])); end);
	#
	if not IsEqualSet(originalposRspol, dones) then Printn("wrong in Make_PosOrthogonalRoots 2"); Error(); fi;
	if rtnumb<>2*Length(originalposRspol) then Printn("wrong in Make_PosOrthogonalRoots 3"); Error(); fi;
	#
	return(data);
	#
end;

HoloSphereLattice:=function(Gram, f)
	#
	# Let Gram be a Gram matrix of a hyperbolic lattice (signature (1, rank-1))
	# and f is a (non-zero) primitive vector such that f^2=0.
	# This function returns a record HSrec on the structure of
	# the negative-definite lattice  <f>^{\perm}/<f>:
	#
	# HSrec.basis: a list of vectors in <f>^{\perm} whose projection to <f>^{\perm}/<f>
	# form a bsis of <f>^{\perm}/<f>.
	# HSrec.Gram: the Gram matrix of <f>^{\perm}/<f> with respect to this basis.
	#
	#
	local Gf, hnfrec, ff;
	Gf:=TransposedMat([Gram*f]);
	hnfrec:=HermiteNormalFormIntegerMatTransform(Gf);
	ff:=hnfrec.rowtrans[1];
	return(OrthogonalComp([f,ff], Gram));
end;

RootType:=function(rank, disc)
	local name, names;
	names:=[];
	for name in RecNames(ADEHrec) do
		if ADEHrec.(name).rank=rank and ADEHrec.(name).disc=disc then
			Add(names, name);
		fi;
	od;
	if Length(names)<>1 then
		Printn("wrong in RootType", rank, disc);
		Printn("names", names);
		Error();
	fi;
	return(names[1]);
end;

newRootType:=function(mcGram)
	#
	# mcGram is a positive-definite indecomp ADE-matrix.
	#
	local rk, disc, tname, stdGram, isomperm;
	rk:=Length(mcGram);
	disc:=DeterminantIntMat(mcGram);
	tname:=RootType(rk, disc);
	stdGram:=ADEHrec.(tname).Gram;
	isomperm:=FindASubconfig(stdGram, mcGram);
	if isomperm=false or isomperm=fail then
		Printn("wrong in newRootType"); Error();
	fi;
	return([tname, isomperm]);
end;

affineRootType:=function(mcGram) # minus of the Gram of the component
	local name, rank, stdGram, isomperm;
	rank:=Length(mcGram)-1;
	for name in RecNames(affineADErec) do
		if affineADErec.(name).rank=rank then
			stdGram:=affineADErec.(name).Gram;
			isomperm:=FindASubconfig(stdGram, mcGram);
			if isomperm<>false and isomperm<>fail  then
				return([name, isomperm]);
			fi;
		fi;
	od;
	Printn("wrong in affineRootType", rank); Error();
	return(fail);
end;


_MTCsFromMTVs:=function(GramS, ample, mtvs)
	#
	# MTVs is the list of all (-2)-vectors  of
	# divisors effective or anti-effective
	# in a sublattice that is generated by MTC, where
	# MTCs is the subset of classes of (-2)-curves,
	# that is the classes of smooth rational curves.
	#
	local ampledual, beep, degvss, tdegs, posvss, tlist,
	irredss, isirred, tpos, tv, newirreds, irreds;
	#
	beep:=function(n)
		Printn("MTCsFromMTVs", n); Error();
	end;
	#
	ampledual:=ample*GramS;
	degvss:=ClassifyByTags(mtvs, tv->tv*ampledual);
	#
	#ClassifyByTags:=function(list, tagfunc)  in gaptools.txt
	#
	tdegs:=List(degvss, xx->xx[1]);
	#
	if not IsSortedList(tdegs) then beep(0); fi;
	posvss:=[]; #this pos is positive
	for tlist in degvss do
		if tlist[1]>0 then Add(posvss, tlist[2]);fi;
		if tlist[1]=0 then beep(1); fi;
	od;
	#
	irredss:=[];
	#
	isirred:=function(v, pos) # this pos is position
		local flag, vdual,ii, tv;
		vdual:=v*GramS;
		flag:=true;
		for ii in [1..pos-1] do
			for tv in irredss[ii] do
				if tv*vdual<0 then
					flag:=false; break;  # from for tv in irredss[ii]
				fi;
			od;
			if not flag then break; fi; # from  for ii in [1..pos-1] do
		od;
		return(flag);
	end;
	#
	for tpos in [1..Length(posvss)] do
		newirreds:=[];
		for tv in posvss[tpos] do
			if isirred(tv, tpos) then
				Add(newirreds, tv);
			fi;
		od;
		#
		Add(irredss, newirreds);
	od;
	#
	irreds:=Concatenation(irredss);
	if RankMat(irreds)<>RankMat(mtvs) then beep(21); fi;
	#
	return(irreds);
end;





RootTypeFromIrredCurves:=function(GramS, irredss)
	#
	local beep, connectedcomps, irredv, newcomp,
	irredvdual, newconnectedcomps, comp, connflag,
	irredv2, intnumb, ii,cr,cdisc,rtype,newvs, name,
	compiidual, jj, tvs, isomperm, cGram, stdGram, mGramS;
	#
	beep:=function(n)
		Printn("wrong RootTypeFromIrredCurves", n); Error();
	end;
	#
	connectedcomps:=[];
	for irredv in irredss do
		newcomp:=[irredv];
		irredvdual:=irredv*GramS;
		newconnectedcomps:=[];
		for comp in connectedcomps do
			connflag:=false;
			for irredv2 in comp do
				intnumb:=irredvdual*irredv2;
				if intnumb>0 then
					connflag:=true; break; # from for irredv2 in comp
				elif intnumb<0 then beep(25);
				fi;
			od;
			if connflag then
				Append(newcomp, comp);
			else
				Add(newconnectedcomps, comp);
			fi;
		od;
		Add(newconnectedcomps, newcomp);
		connectedcomps:=newconnectedcomps;
		Unbind(newconnectedcomps);
	od;
	if Sum(List(connectedcomps, Length))<>Length(irredss) then beep(3); fi;
	#
	for ii in [1..Length(connectedcomps)] do
		compiidual:=TransposedMat(connectedcomps[ii]*GramS);
		for jj in [ii+1..Length(connectedcomps)] do
			if not IsZeroMat(connectedcomps[jj]*compiidual) then
				beep(7);
			fi;
		od;
	od;
	#
	tvs:=[];
	mGramS:=(-1)*GramS;
	for comp in connectedcomps do
		cr:=Length(comp);
		cdisc:=DeterminantMat(TMTTmult(comp, mGramS));
		name:=RootType(cr, cdisc);
		stdGram:=(-1)*ADEHrec.(name).Gram;
		cGram:=TMTTmult(comp, GramS);
		isomperm:=FindASubconfig(stdGram, cGram); # not cGram, stdGram !!
		if isomperm=false or isomperm=fail then beep(10); fi;
		newvs:=List(isomperm, ii->comp[ii]);
		if TMTTmult(newvs, GramS)<>stdGram then beep(20); fi;
		Add(tvs, [name, newvs]);
	od;
	#
	Sort(tvs, function(xx, yy) return(indecompADEsorter(xx[1], yy[1])); end);
	#
	return(tvs);
end;

GeomRootType:=function(GramS, h, ample)
	#
	# GramS is an intersection matrix of signature (1, r-1),
	# h is a polarization with h^2>0
	# and ample is an ample polarization.
	# The output is the list of [type, the list of h-exceptional classes].
	# the list of h-exceptional classes is given
	# in such a way that the Gram matrix of h-exceptional classes
	# is as ADEHrec.
	#
	#
	local beep, mtvs, irreds, tvs;
	#
	beep:=function(n)
		Printn("wrong GeomRootType", n); Error();
	end;
	#
	mtvs:=AffESstd(GramS, h, 0, -2, true);
	if mtvs=[] then
		return([0, []]);
	fi;
	irreds:=_MTCsFromMTVs(GramS, ample, mtvs);
	#
	#
	tvs:=RootTypeFromIrredCurves(GramS, irreds);
	# already sorted
	#
	return(tvs);
	#
end;

RatInvolutionMat:=function(GramS, th, pullbacks, geomroots, checklevel)
	#
	# th is a polarization that induces an involution.
	# pullbacks is a vector of the pull-back of the quotient rational surface.
	# If quot is P^2, then [the pull-back of a line].
	# For an elliptic inversion,
	# th:= 3*fiber+zerosection;
	# pullbacks:=[fiber, zerosection];
	#
	local tcomp,  tk, invariantvs, rt, tn, ade, excepvs, beep,
	antiinvariantvs, diminv, dimantiinv, M, Msprime, orthorec, E6inv, S;
	#
	beep:=function(n)
		Printn("wrong RatInvolutionMat", n); Error();
	end;
	#
	invariantvs:=List(pullbacks);
	#
	if geomroots<>[ 0, [  ] ] then
		for tcomp in geomroots do
			rt:=tcomp[1];
			tn:=ADEHrec.(rt).rank;
			ade:=ADEHrec.(rt).type;
			excepvs:=tcomp[2];
			if ade='A' then
				if tn mod 2 =1 then
					Add(invariantvs, excepvs[(tn+1)/2]);
				fi;
				for tk in [1..FloorRat(tn/2)] do
					Add(invariantvs, excepvs[tk]+excepvs[tn+1-tk]);
				od;
			elif ade='D' then
				if tn mod 2=0 then
					Append(invariantvs, excepvs);
				else
					Add(invariantvs, excepvs[1]+excepvs[2]);
					Append(invariantvs, List([3..tn], kk->excepvs[kk]));
				fi;
			elif ade='E' then
				if tn=6 then
					E6inv:=[excepvs[1], excepvs[4], excepvs[2]+excepvs[6], excepvs[3]+excepvs[5]];
					Append(invariantvs, E6inv);
				elif tn=7 then
					Append(invariantvs, excepvs);
				elif tn=8 then
					Append(invariantvs, excepvs);
				fi;
			else
				beep(0);
			fi;
		od;
	fi;
	#
	if RankMat(invariantvs)<>Length(invariantvs) then beep(1); fi;
	#
	orthorec:=OrthogonalComp(invariantvs, GramS);
	#
	antiinvariantvs:=orthorec.basis;
	diminv:=Length(invariantvs);
	dimantiinv:=Length(antiinvariantvs);
	S:=StackMats(invariantvs, antiinvariantvs);
	#
	if dimantiinv>0 then #Corrected 2022/03/03
		Msprime:=DiagonalMats([IdentityMat(diminv), (-1)*IdentityMat(dimantiinv)]);
	else
		Msprime:=IdentityMat(diminv);
	fi;
	M:=InverseMat(S)*Msprime*S;
	#
	if checklevel>0 then
		if not IsIntMat(M) then beep(2); fi;
		if M*M<>IdentityMat(Length(GramS)) then beep(3); fi;
		if TMTTmult(M, GramS)<>GramS then beep(4); fi;
		if (th*M*GramS)*th<=0 then beep(5); fi;
		if pullbacks<>pullbacks*M then beep(6); fi;
		if geomroots<>[ 0, [  ] ] then
			for tcomp in geomroots do
				if not IsEqualSet(tcomp[2], tcomp[2]*M) then beep(7); fi;
			od;
		fi;
	fi;
	if checklevel>1 then
		if SeparatingVects(GramS, th, th*M, -2) <>[] then beep(8); fi;
	fi;
	#
	return(M);
	#
end;

SeparatingRatVects:=function(GramS, pol, v, d)
	# GramS is a Gram matrix of a hyperbolic lattice.
	# pol and v are RATIONAL vectors of positive square norm in the same positive cone.
	#
	# This function returns the list of all [ri, ti] such that ri is a vector
	# in the lattice GramS satisfying
	# (ri, pol)>0, (ri, v)<0,  (ri, ri)=d, and
	# and (pol+ti*v, ri)=0, where ti are positive real numbers.
	# When pol is pol, the input d=-2 gives a criterion for nef.
	#
	local intpol, intv, beep, intpoldata, intvdata,
	apol, av, rescale, tsepvs, sepvs,trt, tr, trdual,
	tt, newtt;
	#
	beep:=function(numb)
		Printn("wrong SeparatingRatVects", numb); Error();
	end;
	#
	intpoldata:=MakePrimitiveRatVect(pol);
	intvdata:=MakePrimitiveRatVect(v);
	apol:=intpoldata[1];
	av:=intvdata[1];
	if apol<=0 then beep(-1); fi;
	if av<=0 then beep(-2); fi;
	intpol:=intpoldata[2];
	intv:=intvdata[2];
	if apol*intpol<>pol then beep(1); fi;
	if av*intv<>v then beep(2); fi;
	rescale:=apol/av;
	tsepvs:=SeparatingVects(GramS, intpol, intv, d); #Ints!
	sepvs:=[];
	for trt in tsepvs do
		tr:=trt[1];
		trdual:=tr*GramS;
		if tr*trdual<>d then beep(34567); fi;
		tt:=trt[2];
		newtt:=rescale*tt;
		if trdual*(pol+newtt*v)<>0 then
			beep(54738);

MakeEnriquesRec:=function(GramSX, tg)
	#
	# GramSX is the Gram matrix of the covering K3.
	# tg is an involution.
	#
	local beep, Idr, emb, GramSSY, r, GramSY, GramSXdual,
	PSSYSX, PSSYSXG, Enrrec, minusfours, embSSYtoSX,
	projSXdualtoSSYdual, projSXtoSSY, embSSYdualtoSXdual, MinusFoursInSX,
	Id10, GramSYdual, IdP, GramP, GramPdual, projSXdualtoPdual, projSXtoP,
	embPdualtoSXdual, embPtoSX;
	#
	beep:=function(n)
		Printn("wrong in MakeEnrquesRec", n); Error();
	end;
	#
	r:=Length(GramSX);
	GramSXdual:=InverseMat(GramSX);
	Idr:=IdentityMat(r);
	Id10:=IdentityMat(10);
	IdP:=IdentityMat(r-10);
	#
	# for check
	if not IsInvol(GramSX, tg) then beep(1); fi;
	#
	emb:=OrthogonalComp(TransposedMat(tg-Idr),  Idr);
	#
	# for check
	if not emb.basis*tg=emb.basis then
		beep(2);
	fi;
	#
	if Length(emb.basis)<>10 then return(false); fi;
	GramSSY:=TMTTmult(emb.basis, GramSX);
	#
	# for check
	if SignatureQ(GramSSY)<>[10, 1, 9] then
		beep(4);
	fi;
	#
	GramSY:=(1/2)*GramSSY; #(1/2) is importatnt.
	if not IsEvenLattice(GramSY) then return(false); fi;
	if DeterminantMat(GramSY)<>-1 then return(false); fi;
	GramSYdual:=InverseMat(GramSY);
	#
	PSSYSX:=OrthogonalComp(emb.basis, GramSX);
	GramP:=PSSYSX.Gram;
	if ShortestVectors((-1)*GramP, 2).vectors<>[] then
		# No root condition
		return(false);
	fi;
	minusfours:=ShortestVectors((-1)*GramP, 4).vectors;
	minusfours:=Concatenation(minusfours, (-1)*minusfours);
	#
	embSSYtoSX:=emb.basis;
	projSXdualtoSSYdual:=TransposedMat(embSSYtoSX);
	projSXtoSSY:=GramSX*projSXdualtoSSYdual*InverseMat(GramSSY);
	embSSYdualtoSXdual:=TransposedMat(projSXtoSSY);
	#
	GramPdual:=InverseMat(GramP);
	embPtoSX:=PSSYSX.basis;
	projSXdualtoPdual:=TransposedMat(embPtoSX);
	projSXtoP:=GramSX*projSXdualtoPdual*GramPdual;
	embPdualtoSXdual:=TransposedMat(projSXtoP);
	#
	if TMTTmult(embSSYtoSX, GramSX)<>GramSSY then  beep(8822); fi;
	if TMTTmult(embSSYdualtoSXdual, GramSXdual)<>(1/2)*GramSYdual then beep(3); fi;
	if embSSYtoSX*projSXtoSSY<>Id10 then beep(4); fi;
	if embSSYdualtoSXdual*projSXdualtoSSYdual<>Id10 then beep(5); fi;
	#
	if TMTTmult(embPtoSX, GramSX)<>GramP then  beep(88211); fi;
	if TMTTmult(embPdualtoSXdual, GramSXdual)<>GramPdual then beep(31); fi;
	if embPtoSX*projSXtoP<>IdP then beep(41); fi;
	if embPdualtoSXdual*projSXdualtoPdual<>IdP then beep(51); fi;
	#
	MinusFoursInSX:=minusfours*embPtoSX;
	#
	if not IsZeroMat(embPtoSX*projSXtoSSY) then beep(60); fi;
	if not IsZeroMat(embSSYtoSX*projSXtoP) then beep(61); fi;
	if not IsZeroMat(embPtoSX*GramSX*TransposedMat(embSSYtoSX)) then beep(62); fi;
	#
	Enrrec:=rec();
	Enrrec.invol:=tg;
	Enrrec.embSSYtoSX:=embSSYtoSX;
	Enrrec.embSSYdualtoSXdual:=embSSYdualtoSXdual;
	Enrrec.projSXtoSSY:=projSXtoSSY;
	Enrrec.projSXdualtoSSYdual:=projSXdualtoSSYdual;
	Enrrec.GramSX:=GramSX;
	Enrrec.GramSSY:=GramSSY;
	Enrrec.GramSY:=GramSY;
	Enrrec.GramSYdual:=GramSYdual;
	#
	Enrrec.GramP:=GramP;
	Enrrec.GramPdual:=GramPdual;
	Enrrec.embPdualtoSXdual:=embPdualtoSXdual;
	Enrrec.embPtoSX:=embPtoSX;
	Enrrec.projSXtoP:=projSXtoP;
	Enrrec.projSXdualtoPdual:=projSXdualtoPdual;
	#
	Enrrec.MinusFoursInP:=minusfours;
	Enrrec.MinusFoursInSX:=MinusFoursInSX;
	#
	return(Enrrec);
end;

FindRootTypeFromRoots:=function(GramS, rs)
	# rs form a fundamental root system
	local beep, conncomps, ii, newconncomps, intns,
	newcomp, oldcomp, rsdual, bigM, rootnumb,
	adetypes, comp, rr, compvs, compGram, disc, roots;
	#
	#
	beep:=function(numb)
		Printn("wrong FindRootTypeFromRoots", numb); Error();
	end;
	#
	if rs=[] then return([]);fi;
	#
	if Length(rs)<>Rank(rs) then
		beep(9191911);
	fi;
	rsdual:=rs*GramS;
	bigM:=rs*TransposedMat(rsdual);
	conncomps:=[];
	for ii in [1..Length(rs)] do
		newconncomps:=[];
		intns:=bigM[ii];
		newcomp:=[ii];
		for oldcomp in conncomps do
			if 1 in List(oldcomp, jj->intns[jj]) then
				Append(newcomp, oldcomp);
			else
				Add(newconncomps, oldcomp);
			fi;
		od;
		Add(newconncomps, newcomp);
		conncomps:=newconncomps;
	od;
	if not IsDuplicateFree(Flat(conncomps)) then beep(14028); fi;
	if not IsEqualSet(Flat(conncomps), [1..Length(rs)]) then beep(14035); fi;
	#
	adetypes:=[];
	for comp in conncomps do
		rr:=Length(comp);
		compvs:=List(comp, ii->rs[ii]);
		compGram:=TMTTmult(compvs, GramS);
		disc:=AbsInt(DeterminantIntMat(compGram));
		roots:=IntVectsQFstd((-1)*compGram, MakeZeroVect(rr), -2, true);
		rootnumb:=Length(roots);
		Add(adetypes, FindRoottype(rr, disc, rootnumb));
	od;
	#
	return(adetypes);
end;

MakeNefs:=function(GramS, posvect, ample, size)
	#
	local beep, rho, refl, tL, cc, kk, vv, flag, tvv,
	tt, sepssorter, seps, nefs, tp, rt, counter;
	#
	beep:=function(numb)
		Printn("wrong in MakeNefs", numb); Error();
	end;
	#
	rho:=Length(GramS);
	if posvect*GramS*posvect<=0 then
		beep(36538685);
	fi;
	#
	refl:=function(x, r)
		return(x+(x*GramS*r)*r);
	end;
	#
	sepssorter:=function(rt1, rt2)
		local t1, t2, r1, r2;
		t1:=rt1[2];
		t2:=rt2[2];
		if t1>t2 then return(true);
		elif t1<t2 then return(false);
		else
			r1:=rt1[1];
			r2:=rt2[1];
			if r1>r2 then return(true);
			elif r1<r2 then return(false);
			else
				beep(351136);
			fi;
		fi;
		beep(41747141);
	end;
	#
	tL:=[-1,0,1];
	cc:=0;
	kk:=2;
	nefs:=[];
	while Length(nefs)<size do
		cc:=cc+1;
		if cc mod 100 =0 then
			Printn("in MakeNefs", Length(nefs));
			Append(tL, [-kk, kk]);
			kk:=kk+1;
		fi;
		vv:=RandomVectfromL(rho, tL);
		flag:=false;
		tvv:=vv;
		while not flag do
			if tvv*GramS*tvv<=0 then
				tvv:=tvv+posvect;
			else
				flag:=true;
			fi;
		od;
		MakePrimitiveVect(tvv);
		seps:=SeparatingVects(GramS, ample, tvv, -2);
		counter:=0;
		while seps<>[] and counter<10 do
			if counter>1 then Printn("counter", counter, Length(nefs)); fi;
			Sort(seps, sepssorter);
			tvv:=List(tvv);
			for rt in seps do
				tvv:=refl(tvv, rt[1]);
			od;
			MakePrimitiveVect(tvv);
			seps:=SeparatingVects(GramS, ample, tvv, -2);
			counter:=counter+1;
		od;
		if seps=[] then
			Add(nefs, tvv);
		else
			Printn("seps counter", counter, Length(seps));
		fi;
	od;
	nefs:=Set(nefs);
	Printn("Length(nefs)", Length(nefs));
	for tt in [1..size] do
		tp:=RandomChooseFromL(nefs, 2);
		seps:=SeparatingVects(GramS, tp[1], tp[2], -2);
		if seps<>[] then beep(474764646); fi;
	od;
	return(nefs);
end;

			beep(91122);

EasyRandomAmple:=function(Gram, vseed, scale)
  local leng, cc, tv, ttv, ss;
  leng:=Length(vseed);
  tv:=vseed;
  if tv*Gram*tv<=0 then beep(146156); fi;
  cc:=0;
  while cc<10 do
    if IsAmple(Gram, tv) then return(tv); fi;
    ttv:=scale*tv+RandomVectfromL(leng, [-1,1]);
    ss:=scale;
    while ttv*Gram*ttv<=0 do
      ttv:=ss*tv+RandomVectfromL(leng, [-1,1]);
      ss:=ss+1;
      if ss>1000 then beep(69816891); fi;
    od;
    tv:=ttv;
    cc:=cc+1;
  od;
  return(fail);
end;

NewRatInvolutionMat:=function(GramS, th, pullbacks, geomroots, checklevel)
	#
	# th is a polarization that induces an involution.
	# pullbacks is a vector of the pull-back of the quotient rational surface.
	# If quot is P^2, then [the pull-back of a line].
	# For an elliptic inversion,
	# th:= 3*fiber+zerosection;
	# pullbacks:=[fiber, zerosection];
  # geomroots should be prepared by
  # GeomRootType(GramS, th, tample);
	#
	local tcomp,  tk, invariantvs, rt, tn, ade, excepvs, beep,
	antiinvariantvs, diminv, dimantiinv, M, Msprime, orthorec, E6inv, S;
	#
	beep:=function(n)
		Printn("wrong NewRatInvolutionMat", n); Error();
	end;
	#
	invariantvs:=List(pullbacks);
	#
	if geomroots<>[ 0, [  ] ] then
		for tcomp in geomroots do
			rt:=tcomp[1];
			tn:=ADEHrec.(rt).rank;
			ade:=ADEHrec.(rt).type;
			excepvs:=tcomp[2];
			if ade='A' then
				if tn mod 2 =1 then
					Add(invariantvs, excepvs[(tn+1)/2]);
				fi;
				for tk in [1..FloorRat(tn/2)] do
					Add(invariantvs, excepvs[tk]+excepvs[tn+1-tk]);
				od;
			elif ade='D' then
				if tn mod 2=0 then
					Append(invariantvs, excepvs);
				else
					Add(invariantvs, excepvs[1]+excepvs[2]);
					Append(invariantvs, List([3..tn], kk->excepvs[kk]));
				fi;
			elif ade='E' then
				if tn=6 then
					E6inv:=[excepvs[1], excepvs[4], excepvs[2]+excepvs[6], excepvs[3]+excepvs[5]];
					Append(invariantvs, E6inv);
				elif tn=7 then
					Append(invariantvs, excepvs);
				elif tn=8 then
					Append(invariantvs, excepvs);
				fi;
			else
				beep(0);
			fi;
		od;
	fi;
	#
	if RankMat(invariantvs)<>Length(invariantvs) then beep(1); fi;
	#
	orthorec:=OrthogonalComp(invariantvs, GramS);
	#
	antiinvariantvs:=orthorec.basis;
	diminv:=Length(invariantvs);
	dimantiinv:=Length(antiinvariantvs);
	S:=StackMats(invariantvs, antiinvariantvs);
	#
	if dimantiinv>0 then
		Msprime:=DiagonalMats([IdentityMat(diminv), (-1)*IdentityMat(dimantiinv)]);
		M:=InverseMat(S)*Msprime*S;
	else
		Printn("the involution acts trivially on SX");
		M:=IdentityMat(diminv);
	fi;
	#
	if checklevel>0 then
		if not IsIntMat(M) then beep(2); fi;
		if M*M<>IdentityMat(Length(GramS)) then beep(3); fi;
		if TMTTmult(M, GramS)<>GramS then beep(4); fi;
		if (th*M*GramS)*th<=0 then beep(5); fi;
		if pullbacks<>pullbacks*M then beep(6); fi;
		if geomroots<>[ 0, [  ] ] then
			for tcomp in geomroots do
				if not IsEqualSet(tcomp[2], tcomp[2]*M) then beep(7); fi;
			od;
		fi;
	fi;
	if checklevel>1 then
		if SeparatingVects(GramS, th, th*M, -2) <>[] then beep(8); fi;
	fi;
	#
	return(M);
	#
end;

MoveToTheSameChamber:=function(Gram, vwillbemoved, vtarget)
  #
  local beep, seprts, vmoved, movingg, rank, seprtssorter,
  idr, rt, reflmat, vtargetdual, xx;
  #
  beep:=function(beepnumb)
    localbeep("MoveToTheSameChamber", beepnumb); Error();
  end;
  #
  rank:=Length(Gram);
  idr:=IdentityMat(rank);
  reflmat:=function(rv)
    local tv, rvdual;
    rvdual:=rv*Gram;
    return(List(idr, tv->tv+(tv*rvdual)*rv));
  end;
  #
  vtargetdual:=vtarget*Gram;
  seprts:=SeparatingVects(Gram, vwillbemoved, vtarget, -2);
  seprtssorter:=List(seprts, xx->[xx[2], -xx[1]/(xx[1]*vtargetdual)]);
  #
  # Here we use infinitesimal perturbation.
  #
  #Printn("seprts", Length(seprts), "ts", Length(Set(seprts, xx->xx[2])));
  SortParallel(seprtssorter, seprts);
  #
  movingg:=IdentityMat(rank);
  for rt in seprts do
    movingg:=movingg*reflmat(rt[1]);
  od;
  #
  if not IsIntMat(movingg) then beep(691698); fi;
  if TMTTmult(movingg, Gram)<>Gram then beep(57816781); fi;
  #
  vmoved:=vwillbemoved*movingg;
  if SeparatingVects(Gram, vmoved, vtarget, -2)<>[] then
    beep(5751761);
  fi;
  return(movingg);
end;


OrthoRoots:=function(Gram, vs)
  local orthorec, svs,roots, tv, beep;
  beep:=function(beepnumb)
    localbeep("OrthoRoots", beepnumb); Error();
  end;
  orthorec:=OrthogonalCompRec(Gram, vs);
  svs:=ShortestVectors((-1)*orthorec.Gram, 2).vectors;
  if svs=[] then return([]);fi;
  roots:=svs*orthorec.basis;
  roots:=Union(roots, -roots);
  if Set(roots, tv->tv*Gram*tv)<>[-2] then beep(81112); fi;
  return(roots);
end;

DoesKeepTheNef:=function(GramS, amples, tg)
  local amplestg, mindeg, atg, atgdual, aa, ddeg, minpair;
  amplestg:=amples*tg;
  mindeg:=infinity;
  for atg in amplestg do
    atgdual:=atg*GramS;
    for aa in amples do
      ddeg:=atgdual*aa;
      if ddeg<mindeg then
        mindeg:=ddeg;
        minpair:=[atg, aa];
      fi;
    od;
  od;
  return(SeparatingVects(GramS, minpair[1], minpair[2], -2)=[]);
end;
