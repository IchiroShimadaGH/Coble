Make_nRvRs:=function(tGramRdual)
	#
	# make the list of [nR, the list of int vects in tGramRdual with sqnorm nR],
	# where -2 < nR <= 0.
	#
	local svrec;
	svrec:=ShortestVectors((-1)*tGramRdual, 2);
	if Length(svrec.vectors)=0 then Printn("wrong in Make_nRvRs"); Error(); fi;
	return(ClassifyByTags(MakeFullShortestVectors(tGramRdual, svrec), vv->vv*tGramRdual*vv));
end;

Make_nRaRembvRs:=function(tw)
	#
	# make the list of [nR, the list [aR, the list of embL(v)]] (tree),
	# where v is a vector in Rdual such that
	# sqnorm is nR and reldeg with repect to tw is aR.
	#
	# used in Make_navSdata() below.
	#
	local twR, twdeg, ttlist, nRvRs, tlist, aRvss, aRvs;
	#
	twR:=tw*Lrec.proj2;
	twdeg:=function(vv) return(vv*twR); end;
	# vv will be elements of GramRdual.
	#
	ttlist:=[];
	for nRvRs in thenRvRs do
		aRvss:=ClassifyByTags(List(nRvRs[2]), twdeg);
		for aRvs in aRvss do #aRvs[2] is an element of Rdual
			Apply(aRvs[2], vv->vv*Lrec.emb2_dual);
		od;
		Add(ttlist, [nRvRs[1], aRvss]);
	od;
	return(ttlist);
end;


check_nRaRembvRs:=function(tw, tnRaRembvRslist)
	#
	# tnRaRembvRslist is made by Make_nRaRembvRs.
	# used in Make_navSdata(), after Make_nRaRembvRs.
	#
	local beep, svrec, tlist, t2list, navs, nR, avs, aR, v, twdual;
	#
	beep:=function(beepn) Printn("wrong in check_nRaRembvRs", beepn);Error(); end;
	#
	svrec:=ShortestVectors((-1)*GramRdual, 2);
	tlist:=MakeFullShortestVectors(GramRdual, svrec);
	Apply(tlist, vv->vv*Lrec.emb2_dual);
	t2list:=[];
	twdual:=tw*GramL;
	for navs in tnRaRembvRslist do
		nR:=navs[1];
		for avs in navs[2] do
			aR:=avs[1];
			for v in avs[2] do
				if v*GramL*v<>nR then beep(1); fi;
				if twdual*v<>aR then beep(2); fi;
				Add(t2list, CopyVect(v));
			od;
		od;
	od;
	if not IsEqualSet(tlist, t2list) then beep(0); Error(); fi;
	if printlevel>1 then
		Printn("check_nRaRembvRs done: this should be removed");
	fi;
	return(true);
end;

navSdata_sorter:=function(a, b)
	#
	# The order of navSdata should be UNIQUE.
	# tags are all distinct by the definition.
	#
	local taga, tagb;
	taga:=a[1]; tagb:=b[1];
	if taga[5]<tagb[5] then return(true); # number of vectors in SXdual
	elif taga[5]>tagb[5] then return(false);
	else
		if taga[1]<tagb[1] then return(true);  # nS
		elif taga[1]>tagb[1] then return(false);
		else
			if taga[2]<tagb[2] then return(true); # aS
			elif taga[2]>tagb[2] then return(false);
			else
				if taga[3]<tagb[3] then return(true);
				# gcd of the components of a vector in SXdual
				elif taga[3]>tagb[3] then return(false);
				else
					if taga[4]<tagb[4] then return(true);
					# number of rL for each vector in SXdual
					elif taga[4]>tagb[4] then return(false);
					else
						Printn("in navSdata_sorter:");
						## tags are all distinct by the definition.
						Error();
					fi;
				fi;
			fi;
		fi;
	fi;
	return(true);
end;


check_navSdata:=function(navSdata, tw)
	local beep, twdual, twS, navS, nS, aS, gcd, nopslift,
	nopsvect, rSrLs, rSdual, rL, tnavSdata;
	#
	beep:=function(beepn) Printn("wrong in check_navSdata", beepn);Error(); end;
	#
	twdual:=tw*GramL;
	twS:=tw*Lrec.proj1;
	for navS in navSdata do
		nS:=navS[1][1];
		aS:=navS[1][2];
		gcd:=navS[1][3];
		nopslift:=navS[1][4];
		nopsvect:=navS[1][5];
		if Length(navS[2])<>nopsvect then beep(1); fi;
		if nS>=0 then beep(2); fi;
		for rSrLs in navS[2] do
			rSdual:=rSrLs[1];
			if gcd<>Lgcd(rSdual) then beep(3); fi;
			if nopslift<>Length(rSrLs[2]) then beep(4); fi;
			if rSdual*GramSXdual*rSdual<>nS then beep(8); fi;
			if rSdual*twS<>aS then beep(9); fi;
			for rL in rSrLs[2] do
				if rL*GramL*rL<>-2 then beep(5); fi;
				if rL*twdual<>1 then beep(6); fi;
				if rL*Lrec.proj1_dual<>rSdual  then beep(7); fi;
			od;
		od;
	od;
	#
	tnavSdata:=List(navSdata);
	Sort(tnavSdata, navSdata_sorter);
	if tnavSdata<>navSdata then
		beep(10);
	fi;
	if printlevel>1 then
		Printn("check_navSdata done: this should be removed");
	fi;
	return(true);
end;




Make_navSdata:=function(tw)
	#
	#
	# The output is navSdata,
	# a list of navS sorted by navSdata_sorter.
	#
	# Each item navS in navSdata is
	# a pairs of
	# navS[1] =[nS, aS, gcd, nopslift, nopsvect] (called tag)
	# and a list navS[2] of [a vector vS in Sdual, lifts of vS in L].
	# nS:=navS[1][1];
	# aS:=navS[1][2];
	# gcd:=navS[1][3];
	# nopslift:=navS[1][4];
	# nopsvect:=navS[1][5];
	#
	# if Length(navS[2])<>nopsvect then beep(1); fi;
	# See check_navSdata for the meaning.
	#
	#
	#
	local nRaRembvRs, twSdual, nRvRs, embSdualL, navSdata, navs, avs, aR, aS, nrSs,
	arSs, nR, embrRs, nS, tlist, task, tagfunc, addnops_to_tag, tlistwithtags;
	#
	twSdual:=tw*Lrec.proj1_dual;
	embSdualL:=Lrec.emb1_dual;
	nRaRembvRs:=Make_nRaRembvRs(tw);
	if checklevel>0 then
		check_nRaRembvRs(tw, nRaRembvRs);
	fi;
	navSdata:=[];
	#
	for navs in nRaRembvRs do
		nR:=navs[1];
		nS:=-2-nR;
		if nS<0 then
			for avs in navs[2] do
				aR:=avs[1];
				embrRs:=avs[2];
				aS:=1-aR;
				tlist:=[];
				task:=function(trS)
					local tembtrS,  tembrR, trL, trLs;
					tembtrS:=trS*embSdualL;
					trLs:=[];
					for tembrR in embrRs do
						trL:=tembtrS+tembrR;
						if IsIntVect(trL) then
							Add(trLs, trL);
						fi;
					od;
					if trLs<>[] then
						Add(tlist, [CopyVect(trS), trLs]);
					fi;
				end;
				AffES(GramSXdual, twSdual, aS, nS, true, task);
				if Length(tlist)>0 then
					tagfunc:=function(rSrLs)
						return([nS, aS, Lgcd(rSrLs[1]), Length(rSrLs[2])]);
					end;
					tlistwithtags:=ClassifyByTags(tlist, tagfunc);
					addnops_to_tag:=function(taglist)
						taglist[1][5]:=Length(taglist[2]);
						return(taglist);
					end;
					Apply(tlistwithtags, addnops_to_tag);
					Append(navSdata, tlistwithtags);
				fi;
			od;
		fi;
	od;
	Sort(navSdata, navSdata_sorter);
	if checklevel>0 then
		check_navSdata(navSdata, tw);
	fi;
	return(navSdata);
end;

Make_MinimalChunk:=function(navSdata)
	local  bigvs, kk, bigM, basisdual, vv, tn, targetM, minchunkrec;
	#
	#
	bigvs:=List(navSdata[1][2], xx->xx[1]);
	kk:=1;
	while RankMat(bigvs)<rankSX do
		kk:=kk+1;
		Append(bigvs, List(navSdata[kk][2], xx->xx[1]));
	od;
	if RankMat(bigvs)<rankSX then
		Printn("wrong in Make_MinimalChunk"); Error();
	fi;
	bigM:=TMTTmult(bigvs, intGramSXdual);
	basisdual:=[];
	for vv in bigvs do
		Add(basisdual, vv);
		tn:=RankMat(basisdual);
		if tn<Length(basisdual) then
			Remove(basisdual);
		fi;
		if tn=rankSX then break; fi;
	od;
	targetM:=TMTTmult(basisdual, intGramSXdual);
	minchunkrec:=rec(
		numb:=kk,
		targetM:=targetM,
		bigM:=bigM,
		basisdual:=basisdual,
		bigvs:=bigvs
	);
	return(minchunkrec);
end;

NewMake_MinimalChunk:=function(navSdata)
	#
	# 2018/01/05
	# Find a bigvs good for backtrack search.
	#
	local  bigvs, kk, thekk, bigM, basisdual, vv, tn, targetM, minchunkrec,
	tbigvs, hightpoint, points, trial, tbigM, tbasisdual, ttargetM,
	randomminchunktrial, estimatebound, maxpoints, trange, cc, ccrange;
	#
	#
	bigvs:=List(navSdata[1][2], xx->xx[1]);
	kk:=1;
	while RankMat(bigvs)<rankSX do
		kk:=kk+1;
		Append(bigvs, List(navSdata[kk][2], xx->xx[1]));
	od;
	thekk:=kk;
	if RankMat(bigvs)<rankSX then
		Printn("wrong in NewMake_MinimalChunk"); Error();
	fi;
	#
	#
	Reset(GlobalMersenneTwister, BorSeed);
	#
	maxpoints:=MakeZeroVect(Length(bigvs));
	randomminchunktrial:=Length(bigvs);
	estimatebound:=minchunk_aa*rankSX^2;
	for trial in [1..randomminchunktrial] do
		trange:=[1..Minimum(randomminchunktrial-trial+14, Length(bigvs))];
		ccrange:=Length(bigvs)-Length(trange);
		for cc in [1..minchunk_bb*ccrange] do
			if cc>ccrange then
				tbigvs:=PartialShuffle(bigvs, [1..randomminchunktrial]);
			else
				tbigvs:=PartialShuffle(bigvs, [1..ccrange]);
			fi;
			tbigM:=TMTTmult(tbigvs, intGramSXdual);
			tbasisdual:=[];
			for vv in tbigvs do
				Add(tbasisdual, vv);
				tn:=RankMat(tbasisdual);
				if tn<Length(tbasisdual) then
					Remove(tbasisdual);
				fi;
				if tn=rankSX then break; fi;
			od;
			ttargetM:=TMTTmult(tbasisdual, intGramSXdual);
			points:=NewFindAllSubconfigsEstimate(ttargetM, tbigM, estimatebound);
			if points>maxpoints then
				#Printn("Estimate", trial, ccrange, cc>ccrange,  List([1..10], ii->points[ii]), List([1..10], ii->maxpoints[ii]));
				maxpoints:=points;
				bigvs:=tbigvs;
				bigM:=tbigM;
				basisdual:=tbasisdual;
				targetM:=ttargetM;
			fi;
			if points[1]=infinity then
				break; #from for trial in randomminchunktrial do
			fi;
		od;
	od;
	#
	minchunkrec:=rec(
		numb:=thekk,
		targetM:=targetM,
		bigM:=bigM,
		basisdual:=basisdual,
		bigvs:=bigvs
	);
	return(minchunkrec);
end;

DoesPreservePeriod:=function(tg) # tg in OSX
	#
	# This also check whether tg preserves the positive cone.
	# When periodflag is false, it only checks the liftability.
	# When periodflag is"none", it only checks the positive-cone preservation.
	local disctg;
	#
	if ampleXdual*(ampleX*tg)<=0 then return(false); fi;
	#
	if periodflag="none" then return(true); fi;
	disctg:=CopyMatmodDiscg(discSXrec.reps*tg*discSXrec.proj, discSXrec.discg);
	if not periodflag then
		return(disctg in OqSXliftable);
	fi;
	return(disctg in OqSXperiod);
end;


check_AutST:=function(AutSTdata, tweylS)
	#
	#AutSTdata:=[[AutS,AutT],[AutSdual, AutTdual]];
	#
	local beep, G, gs, gt, gsgt, twSdual, gens, tg, orderG;
	#
	beep:=function(beepn) Printn("wrong in check_AutST", beepn);Error(); end;
	#
	if AutSTdata[1][1][1]<>IdmatSX then beep(100); fi;
	if AutSTdata[1][2][1]<>IdmatSX then beep(200); fi;
	#
	for gs in AutSTdata[1][1] do
		if not IsIntMat(gs) then beep(11); fi;
		if tweylS*gs<>tweylS then beep(12); fi;
		if TMTTmult(gs, GramSX)<>GramSX then beep(1); fi;
		for gt in AutSTdata[1][2] do
			gsgt:=gs*gt;
			if not DoesPreservePeriod(gsgt) then beep(3); fi;
			#Add(G, gsgt);
		od;
	od;
	#
	for gt in AutSTdata[1][2] do
		if not IsIntMat(gt) then beep(21); fi;
		if tweylS*gt<>tweylS then beep(22); fi;
		if TMTTmult(gt, GramSX)<>GramSX then beep(2); fi;
	od;
	#
	#
	gens:=Concatenation(AutSTdata[2][1], AutSTdata[2][2]);
	orderG:=Length(AutSTdata[1][1])*Length(AutSTdata[1][2]);
	#
	if AutSTdata[2][1]<>List(AutSTdata[1][1], tg->GramSXdual*tg*GramSX) then
		beep(14);
	fi;
	if AutSTdata[2][2]<>List(AutSTdata[1][2], tg->GramSXdual*tg*GramSX) then
		beep(24);
	fi;
	#
	if orderG<100 then
		if Size(Group(gens))<>orderG then
			beep(4);
		fi;
		gens:=Concatenation(AutSTdata[1][1], AutSTdata[1][2]);
		if Size(Group(gens))<>orderG then
			beep(8);
		fi;
	else
		twSdual:=tweylS*GramSX;
		G:=[];
		for gs in AutSTdata[2][1] do
			if not IsIntMat(GramSX*gs*GramSXdual) then beep(51); fi;
			if twSdual*gs<>twSdual then beep(52); fi;
			if TMTTmult(gs, GramSXdual)<>GramSXdual then beep(5); fi;
			for gt in AutSTdata[2][2] do
				if not IsIntMat(GramSX*gt*GramSXdual) then beep(61); fi;
				if twSdual*gt<>twSdual then beep(62); fi;
				if TMTTmult(gt, GramSXdual)<>GramSXdual then beep(6); fi;
				gsgt:=GramSX*gs*gt*GramSXdual;
				if not (IsIntMat(gsgt) and DoesPreservePeriod(gsgt)) then
					beep(7);
				fi;
				#Add(G, gsgt);
			od;
		od;
		#if not TestIsMatGroup(G, 10*checklevel) then beep(8); fi;
	fi;
	#
	#
	if printlevel>1 then
		Printn("check_AutST done: this should be removed: ", "orderG=", orderG);
	fi;
	return(true);
end;

IsWallMinSubscript:=function(vs, w)
	#
	# this will be used in IsWallBestCoeffs,
	# when IsWallBestCoeffs gives up after too many pivots.
	#
	# We assume that vs is the full rank.
	# Does the hyperplane w*x=0 intersect with the interior of
	# the cone defined by v*x>=0 for all v in vs?
	# This is a problem of linear programming.
	# The answer is yes if and only if the problem
	#       minimize w*x
	#       subject to v*x>=0
	# has unbounded to -infinity,
	# or if and only if there exists x0 such that
	# w*x0<0 and v*x0>=0 for any v in vs,
	# or if and only if w can NOT  be written as
	# a linear combination of some v's in v with
	# positive coefficients.
	# If true, then it returns [true, x0].
	# If false, then it returns [false, [coeffs, v's]].
	#
	local dic, f, n, nonbasics, svsiis, NBinv, #NBinv is nonbasic inv;
	counter, entcand, pos, c, tflag, y,evpos, sol,
	entvarnumb, entpos, ynumb, leavevarnumb, leavey,a,correcty, pivotfunc,
	NB;
	#
	n:=Length(w);
	dic:=CopyMat(vs);
	f:=CopyVect(w);
	#
	# initialize
	#
	svsiis:=FindSubvsWithRank(vs, rankSX);
	if svsiis=false then Printn("cannot apply IsWall", RankMat(vs)); Error(); fi;
	nonbasics:=svsiis[2];
	NBinv:=Inverse(svsiis[1]);
	Apply(dic, v->v*NBinv);
	f:=f*NBinv;
	#
	counter:=0;
	#
	while counter<10*Length(vs) do
		#
		counter:=counter+1;
		#
		# find entering
		#
		entcand:=[];
		pos:=0; # position in nonbasiscs
		for c in f do
			pos:=pos+1;
			if c<0 then
				#
				tflag:=true;
				for y in dic do
					if y[pos]<0 then tflag:=false; break; fi;
				od;
				if tflag then #unbound!
					NBinv:=Inverse(List(nonbasics, xx->vs[xx]));
					evpos:=MakeVectei(rankSX, pos);
					sol:=NBinv*evpos;
					#Printn(counter, "unbound");
					return([true, sol]);
				fi;
				#
				Add(entcand, nonbasics[pos]);
			fi;
		od;
		#
		if Length(entcand)=0 then # bound
			#Printn(counter, "bound");
			return([false, [f, List(nonbasics, ii->vs[ii])]]);
		fi;
		#
		entvarnumb:=Minimum(entcand);
		entpos:=Position(nonbasics, entvarnumb);
		#
		# find leaving
		#
		ynumb:=0;
		for y in dic do
			ynumb:=ynumb+1;
			if y[entpos]<0 then leavevarnumb:=ynumb; leavey:=y; break; fi;
		od;
		#
		# pivot
		#
		nonbasics[entpos]:=leavevarnumb;
		a:=leavey[entpos];
		correcty:=CopyVect(leavey);
		correcty[entpos]:=-1;
		correcty:=(-1/a)*correcty;
		pivotfunc:=function(yy)
			local aa;
			aa:=yy[entpos];
			yy[entpos]:=0;
			yy:=yy+aa*correcty;
			return(yy);
		end;
		Apply(dic, pivotfunc);
		f:=pivotfunc(f);
		#
		#  check
		if counter <0 then
			NB:=List(nonbasics, ii->vs[ii]);
			if dic*NB<>vs then Printn("wrong pivot 1"); Error(); fi;
			if f*NB<>w then Printn("wrong pivot 2"); Error(); fi;
			if printlevel>1 then
				Printn("pivot check done: this should be removed; counter=", counter);
			fi;
		fi;
		#
		#
		#
		#
		#
	od;
	#
	Printn("too many pivots"); Error();
	#
end;



IsWallBestCoeffs:=function(vs, w)
	#
	# We assume that vs is the full rank and that
	# the region defined by w*x>=0 and v*x>=0 for all v in vs
	# has an interior point.
	# Does the hyperplane w*x=0 intersect with the interior of
	# the cone defined by v*x>=0 for all v in vs?
	# This is a problem of linear programming.
	# The answer is yes if and only if the problem
	#       minimize w*x
	#       subject to v*x>=0
	# has unbounded to -infinity,
	# or if and only if there exists x0 such that
	# w*x0<0 and v*x0>=0 for any v in vs,
	# or if and only if w can NOT  be written as
	# a linear combination of some v's in vs with
	# positive coefficients.
	# If true, then it returns [true, x0].
	# If false, then it returns [false, [coeffs, v's]].
	#
	local dic, f, n, nonbasics, svsiis, NBinv,
	#NBinv is nonbasic inv;
	counter, minc,  y,evpos, sol,
	entvarnumb, entpos, ynumb, leavevarnumb, leavey,a,correcty,
	pivotfunc, NB;
	#
	n:=Length(w);
	dic:=CopyMat(vs);
	f:=CopyVect(w);
	#
	# initialize
	#
	svsiis:=FindSubvsWithRank(vs, rankSX);
	if svsiis=false then
		Printn("cannot apply IsWall", RankMat(vs)); Error();
	fi;
	nonbasics:=svsiis[2];
	NBinv:=Inverse(svsiis[1]);
	Apply(dic, v->v*NBinv);
	f:=f*NBinv;
	#
	counter:=0;
	#
	while counter<1000 do
		#
		counter:=counter+1;
		#
		# find entering
		#
		minc:=Minimum(f);
		#
		if minc>=0 then # bound
			return([false, [f, List(nonbasics, ii->vs[ii])]]);
		fi;
		#
		entpos:=Position(f, minc);
		entvarnumb:=nonbasics[entpos];
		#
		# find leaving
		#
		ynumb:=0;
		a:=0;
		for y in dic do
			ynumb:=ynumb+1;
			if y[entpos]<a then
				a:=y[entpos];
				leavevarnumb:=ynumb;
				leavey:=y;
			fi;
		od;
		#
		if a=0 then
			NBinv:=Inverse(List(nonbasics, xx->vs[xx]));
			evpos:=MakeVectei(rankSX, entpos);
			sol:=NBinv*evpos;
			return([true, sol]); #Printn(counter, "unbound");
		fi;
		#
		# pivot
		#
		nonbasics[entpos]:=leavevarnumb;
		correcty:=CopyVect(leavey);
		correcty[entpos]:=-1;
		correcty:=(-1/a)*correcty;
		pivotfunc:=function(yy)
			local newy, aacorrecty;
			aacorrecty:=yy[entpos]*correcty;
			yy[entpos]:=0;
			newy:=yy+aacorrecty;
			return(newy);
		end;
		Apply(dic, pivotfunc);
		f:=pivotfunc(f);
		#
		#  check
		if counter <0 then
			NB:=List(nonbasics, ii->vs[ii]);
			if dic*NB<>vs then Printn("wrong pivot 1"); Error(); fi;
			if f*NB<>w then Printn("wrong pivot 2"); Error(); fi;
			if printlevel>1 then
				Printn("pivot check done: this should be removed; counter=", counter);
			fi;
		fi;
		#
	od;
	#
	Printn("too many pivots");
	return(IsWallMinSubscript(vs, w));
	#
end;


check_Walls:=function(orbits, result)
	#
	#
	local beep, walls, nonwalls, wallsamples,
	nonwallsamples, ii, res,
	tw, pos, newres, x0, c, vs, v, coeffs, ttw;
	#
	beep:=function(beepn) Printn("wrong check_Walls", beepn); Error(); end;
	#
	if Length(result)<>Length(orbits) then
		beep(0);
	fi;
	#
	walls:=[];
	nonwalls:=[];
	wallsamples:=[];
	nonwallsamples:=[];
	ii:=0;
	#
	for res in result do
		ii:=ii+1;
		if res[1] then
			Append(walls, orbits[ii]);
			Add(wallsamples, RandomOp(orbits[ii]));
		else
			Append(nonwalls, orbits[ii]);
			Add(nonwallsamples, RandomOp(orbits[ii]));
		fi;
	od;
	if RankMat(walls)<>rankSX then beep(1); fi;
	#
	for tw in wallsamples do
		pos:=Position(walls, tw);
		Remove(walls, pos);
		#
		# tw is NOT in walls from now
		#
		if RankMat(walls)=rankSX then
			# Note that, if RankMat(walls)<rankSX, then tw is a wall.
			newres:=IsWallBestCoeffs(walls, tw);
			if not newres[1] then beep(2); fi;
			x0:=newres[2];
			if tw*x0>=0 then beep(3); fi;
			for ttw in walls do
				if ttw*x0<0 then beep(4); fi;
			od;
		fi;
		#
		# tw returns to walls
		#
		Add(walls, tw);
	od;
	#
	#
	for tw in nonwallsamples do
		newres:=IsWallBestCoeffs(walls, tw);
		if newres[1] then beep(5); fi;
		coeffs:=newres[2][1];
		vs:=newres[2][2];
		if coeffs*vs<>tw then beep(6); fi;
		for c in coeffs do if c<0 then beep(7); fi; od;
		for v in vs do if not v in walls then beep(8); fi;od;
	od;
	#
	if printlevel>1 then
		Printn("check_Walls done: this should be removed.");
	fi;
	return(true);
end;

FindIsom:=function(oldvectsdualinv, targetM, newvectsdual, bigM, olddefvss, newdefvss)
	#
	local findjob, n, N, partial, globalfflag, isom;
	#
	N:=Length(bigM);
	n:=Length(targetM);
	partial:=MakeZeroVect(n);
	#Printn("FindIsom", N, n); #added 2017 Oct 26
	#
	findjob:=function(kk)
		#
		# When this function is called,
		# partial has kk correct elements from the top;
		# that is, F(partial[ii], partial[jj])=f(ii, jj) for ii, jj<=kk.
		# This function search for kk+1 st element.
		#
		local targetv, jj, bigvjj, fflag, cc, ii, tgdual,
		pos, defvssflag, defvs;
		#
		if kk=n then
			isom:=oldvectsdualinv*List(partial, ii-> newvectsdual[ii]);
			#
			defvssflag:=true;
			pos:=0;
			for defvs in olddefvss do
				pos:=pos+1;
				if not IsEqualSet(defvs*isom, newdefvss[pos]) then
					defvssflag:=false;
					break;
				fi;
			od;
			#
			if defvssflag then
				isom:=GramSX*isom*GramSXdual; #make it non-dual
				if IsIntMat(isom) and DoesPreservePeriod(isom) then
					globalfflag:=true;
				fi;
			fi;
		else
			targetv:=targetM[kk+1];
			for jj in [1..N] do
				#
				bigvjj:=bigM[jj];
				partial[kk+1]:=jj;
				fflag:=true;
				cc:=0;
				for ii in partial do
					cc:=cc+1;
					if bigvjj[ii]<>targetv[cc] then
						fflag:=false; break;
					fi;
					if cc=kk+1 then break; fi;
				od;
				if fflag then
					findjob(kk+1);
				fi;
				if globalfflag then break; fi;
				#
			od;
		fi;
	end;
	#
	globalfflag:=false;
	findjob(0);
	#
	if globalfflag=false then return(false);
	elif globalfflag=true then  return(isom);
	fi;
	#
	Printn("wrong in FindIsom"); Error();
end;

check_Isom:=function(tweyl, isomres, oldweyl)
	#
	# isomres=[oldchamnumb, tg];
	#
	local beep,  oldweylS, tweylS, tg, oldchamnumb;
	beep:=function(beepn) Printn("wrong in check_Isom", beepn); Error(); end;
	#
	oldchamnumb:=isomres[1];
	tg:=isomres[2];
	if TMTTmult(tg, GramSX)<>GramSX then beep(1); fi;
	if not DoesPreservePeriod(tg) then beep(11); fi;
	oldweylS:=oldweyl*Lrec.proj1;
	tweylS:=tweyl*Lrec.proj1;
	if oldweylS*tg<>tweylS then beep(2); fi;
	if printlevel>1 then
		Printn("check_Isom done: this should be removed");
	fi;
	return(true);
end;

IsNewCham:=function(tweyl, newFP, allnewvectsdual, newbigM, newdefvss)
	#tweyl is needed  for check
	local oldFP, isom, isomres, cc, nopsFPs;
	nopsFPs:=Length(FingerPrints);
	for cc in [0..nopsFPs-1] do
		oldFP:=FingerPrints[nopsFPs-cc];
		if newFP.navstag=oldFP.navstag then
			isom:=FindIsom(oldFP.dualbasisinv, oldFP.dualbasisGram,
					allnewvectsdual, newbigM, oldFP.defvss, newdefvss);
			if isom<>false then
				isomres:=[oldFP.chamnumb, isom];
				if checklevel>-1 then # Here -1
			 		check_Isom(tweyl, isomres, oldFP.wdata[1]);
				fi;
				return(isomres);
			fi;
		fi;
	od;
	return(true);
end;

check_enteringwall:=function(enteringwall, torbits, twalls)
	#
	# used in Make_Cham.
	# This check is executed always (without regarding checklevel).
	#
	local beep, fflag, ii, torb;
	#
	beep:=function(beepn) Printn("wrong check_enteringwall", beepn); Error(); end;
	fflag:=false;
	ii:=0;
	for torb in torbits do
		ii:=ii+1;
		if  enteringwall in torb then
			if twalls[ii][1]=false then beep(1); fi;
			fflag:=true;
			break;
		fi;
	od;
	if not fflag then beep(2); fi;
	#Printn("check_enteringwall done:this should be removed");
	return(true);
end;

Borcherds:=function(tchecklevel, tperiodflag, savetab, tsaveweylstab, datadir, AutSTparameters)
	#
	# tchecklevel is 1 if simple Borcherds
	# and 0 if not simple Borcherds.
	#
	# tperiodflag is usually true.
	# the positive-cone preserving is always checked.
	# periodflag is true, false, or "none"
    # true:  period check in DoesPreservePeriod
    # false: liftability check in DoesPreservePeriod
    # "none": only positive-cone preserving
	#
	# savetab is 1 if simple Borcherds
	# and large if not simple Borcherds.
	#
	# tsaveweylstab is 1 if simple Borcherds
	# and large if not simple Borcherds.
	#
	# The defualt is AutSTparameters=[false, infinity, 0, 0]
	# if some AutST is large, one should use NewMake_MinimalChunk
	# by putting some number in BorSeed:=AutSTparameters[1],
	# and putting
	# minchunk_aa:=1 or so;
	# minchunk_bb:=2 or so;
	# check_AutSTnops:=  100 or so;
	#
	#
	local tweylnumb, st, fn;
	#
	checklevel:=tchecklevel;
	periodflag:=tperiodflag;
	saveweylstab:=tsaveweylstab;
	Printn("checklevel:=", checklevel, "periodflag:=", periodflag, "saveweylstab:=", saveweylstab, "adjchecknumb:=", adjchecknumb);
	#
	BorSeed:=AutSTparameters[1];
	check_AutSTnops:=AutSTparameters[2];
	minchunk_aa:=AutSTparameters[3];
	minchunk_bb:=AutSTparameters[4];
	#
	PrepareDataBorcherds(datadir);
	#
	currentreadingweylnumb:=1;
	totalweylnumb:=1;
	currentreadingweylnumb_in_list:=1;
	NEXTweylLISTnumb:=2;
	savedweyllists:=[];
	# the list of the names of the files of savedWeylVectors
	# that are saved in SaveWeylVectors.
	savedchamlists:=[];

	savedWeylVectors:=[];

			st:=Runtime();
			fn:=Filename(_datasdir, Concatenation("savechams", String(tweylnumb), ".txt"));
			PrintTo(fn, "tempChamsData:=", tempChamsData, ";\n");

			SaveWeylVectors();

			PrintTo(fn, "FingerPrints:=", FingerPrints, ";\n");
			time8:=time8+Runtime()-st;
			AutK3Monitor(tweylnumb);
		fi;
		#
		if printlevel>0 then
			Printn("----------------------------", tweylnumb, "done");
		fi;
		tweylnumb:=tweylnumb+1;
	od;
	#
	if tempChamsData<>[] then
		fn:=Filename(_datasdir, Concatenation("savechams", String(totalweylnumb), ".txt"));
		PrintTo(fn, "tempChamsData:=", tempChamsData, ";\n");

		fn:=Filename(_datasdir, Concatenation("saveFingerPrints", ".txt"));

		AutK3Monitor(totalweylnumb);
		#
		Printn("----------------------------", totalweylnumb, "done");
	fi;
	#
