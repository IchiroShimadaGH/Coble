RandFindRankM:=function(lfset, rank, seed)
	local rlfset;
	if seed<>false then 
		Reset(GlobalMersenneTwister, seed);
	fi;
	rlfset:=ShallowCopy(lfset);
	Shuffle(rlfset);
	return(FindRankM(rlfset, rank));
end;

#
# Below, linforms are used to the walls of a chamber.
# No two are proportional,
# and they are full rank (essential).

	

RandFind1dimFace:=function(linforms, seed, trial)
	local beep, n, tM, tt, ii,
	tsols, tsol, aa, negtlfs, tlf, m, fflag2,
	tM2, tM3, tM23, tt2, tlfsminus, tlfsplus, ttM;
	#
	beep:=function(numb)
		Printn("wrong RandFind1dimFace"); Error();
	end;
	#
	if seed<>false then 
		Reset(GlobalMersenneTwister, seed);
	fi;
	n:=Length(linforms[1]);
	if n<=2 then
		beep(1);
	fi;
	#
	tM:=RandFindRankM(linforms, n-1, false);
	tt:=0;
	while tt<trial  do
		tt:=tt+1;
		tsols:=NullspaceMat(TransposedMat(tM));
		if Length(tsols)<>1 then beep(1); fi;
		tsol:=tsols[1];
		tlfsplus:=[];
		tlfsminus:=[];
		for tlf in linforms do
			aa:=tlf*tsol;
			if aa>0 then
				Add(tlfsplus, tlf);
			elif aa<0 then 
				Add(tlfsminus, tlf);
			fi;
		od;
		#if Length(tlfsminus)<=Length(tlfsplus) then
		if Random([-1,1])=1 then
			negtlfs:=tlfsminus;
		else
			tsol:=-tsol;
			negtlfs:=tlfsplus;
		fi;
		if negtlfs=[] then 
			#Printn("___found one-dim face");
			return(tsol); 
		fi;
		m:=Rank(negtlfs);
		if m<n-1 then 
			fflag2:=false;
			for tt2 in [1..trial] do
				tM2:=RandFindRankM(negtlfs, m, false);
				ttM:=ShallowCopy(tM);
				Shuffle(ttM);
				tM3:=List([1..n-1-m], ii->ttM[ii]); 
				tM23:=StackMats(tM2, tM3);
				if Rank(tM23)=n-1 then
					tM:=tM23;
					fflag2:=true;
					break;
				fi;
			od;
			if not fflag2 then
				tM:=RandFindRankM(linforms, n-1, false);
			fi;
		else
			tM:=RandFindRankM(negtlfs, n-1, false);
		fi;
	od;
	#
	#
	#dbglinforms:=linforms;
	#Printn("cannot RandFind1dimFace"); 
	return(fail); 
end;

SignRat:=function(rat)
	return(SignInt(NumeratorRat(rat)));
end;


NormalVolume:=function(M)
	local num, denom;
	num:=Determinant(M)^2;
	denom:=Product(M, xx->xx^2);
	return(num/denom);
end;


RandFindInteriorPoint:=function(linforms, seed, trial)
	local beep, n, interiorpoint, tt, isinterior, tsol,
	onedimfaces, ns, xx, signflags, ii, jj, tlf,
	ttM, irrelevants, relevants, spandim,
	findtrue, findfalse, pos, tB, tBinv, vv, res, counter,
	doneBs, findtruetrial, findfalsetrial,
	newrelevants, isawallflag, rest;
	#
	#
	beep:=function(numb)
		Printn("wrong RandFindInteriorPoint", numb); Error();
	end;
	#
	if seed<>false then 
		Reset(GlobalMersenneTwister, seed);
	fi;
	#
	ns:=Set(linforms, Length);
	if Length(ns)<>1 then beep(22); fi;
	n:=ns[1];
	#
	for ii in [1..Length(linforms)] do
		for jj in [ii+1..Length(linforms)] do
			if Rank([linforms[ii], linforms[jj]])<>2 then
				beep(33);
			fi;
		od;
	od;
	#
	if Rank(linforms)<>n then
		beep(44);
	fi;
	#
	interiorpoint:=MakeZeroVect(n);
	#
	isinterior:=function()
		local tlf, aa;
		for tlf in linforms do
			aa:=tlf*interiorpoint;
			if aa<=0 then beep(4223); fi;
		od;
		return(true);
	end;
	#
	#Add(examples, List(linforms));
	#
	if n=1 then
		signflags:=List(linforms, tlf->SignRat(tlf[1]));
		if Set(signflags)=[1] then
			return([true, [1]]);
		elif Set(signflags)=[-1] then
			return([true, [-1]]);
		else
			if linforms[1][1]>0 then 
				pos:=Position(signflags, -1);
			else
				pos:=Position(signflags, 1);
			fi;
			return([false, [linforms[1]], linforms[pos]]); 
		fi;
	fi;
	#
	if n=2 then
		onedimfaces:=[];
		for tlf in linforms do
			tsol:=[tlf[2], -tlf[1]];
			signflags:=List(linforms, tlf->SignRat(tsol*tlf));
			if NumbOccur(0, signflags)<>1 then beep(34567); fi;
			signflags:=Set(signflags);
			if Length(signflags)<2 then beep(34567); fi;
			if signflags=[0,1] then
				Add(onedimfaces, tsol);
			elif signflags=[-1,0] then
				Add(onedimfaces, -tsol);
			fi;
		od;
		if Length(onedimfaces)<>2 then
			#Printn("no interior point"); 
			for ii in [1..Length(linforms)] do
				for jj in [ii+1..Length(linforms)] do
					tB:=[linforms[ii], linforms[jj]];
					tBinv:=InverseMat(tB);
					for tlf in linforms do
						vv:=tlf*tBinv;
						if not 1 in List(vv, SignRat) then
							return([false, [tlf, vv, tB]]); 
						fi;
					od;
				od;
			od;
		else
			interiorpoint:=Sum(onedimfaces);
			if not isinterior() then
				beep(35679);
			fi;
			return([true, interiorpoint]);
		fi;
	fi;
	#
	####################
	#
	# for n>2
	#
	relevants:=ShallowCopy(linforms);
	irrelevants:=[];
	onedimfaces:=[];
	spandim:=0;
	findtruetrial:=spandim+trial;
	#
	findtrue:=function()
		#
		local tt, tsol;
		#
		tt:=0;
		while tt<findtruetrial  do
			tt:=tt+1;
			tsol:=RandFind1dimFace(relevants, false, findtruetrial);
			if tsol<>fail then
				if Rank(CopyAdd(onedimfaces, tsol))>spandim then
					Add(onedimfaces, tsol);
					spandim:=spandim+1;
					if spandim=n then 
						interiorpoint:=Sum(onedimfaces);
						if not isinterior() then
							beep(3115679);
						fi;
						return(interiorpoint);
					fi;
				fi;
			fi;
		od;
		return(fail);
	end;
	#
	doneBs:=[];
	findfalsetrial:=0;
	findfalse:=function(isthereonedimface)
		#
		local tt, tB, tBinv, pivotcounter, newttM, tlf, vv,
		signflags, pos, mintB, nvol,  newtB, newnvol, trialnumb;
		#
		tt:=0;
		if not isthereonedimface then
			trialnumb:=2*findfalsetrial;
		else
			trialnumb:=findfalsetrial;
		fi;
		while tt<trialnumb  do
			tB:=RandFindRankM(relevants, n, false);
			if not tB in doneBs then
				Add(doneBs, tB);
				tBinv:=InverseMat(tB);
				#Printn("___new pivots", tt, Length(doneBs));
				pivotcounter:=0;
				nvol:=NormalVolume(tB);
				while pivotcounter<trialnumb  do
					newttM:=[]; # from relevants, we remove further irrelevant vectors.
					for tlf in relevants do
						if  tlf in tB then  
							Add(newttM, tlf);
						else
							vv:=tlf*tBinv;
							signflags:=List(vv, SignRat);
							# if all comps of vv are >=0, then tlf is not relevant.
							if not -1 in signflags then
								Add(irrelevants, [tlf, vv, tB]);
							else
								Add(newttM, tlf);
								if not 1 in signflags then
									#if isthereonedimface then
									#	beep(77331122);
									#fi;
									return([tlf, vv, tB]);
								fi;
								mintB:=tB;
								for pos in [1..n] do
									if vv[pos]<>0 then
										newtB:=ShallowCopy(tB);
										newtB[pos]:=tlf;
										newnvol:=NormalVolume(newtB);
										if newnvol<=0 then beep(46728); fi;
										if newnvol<nvol then
											mintB:=newtB;
											nvol:=newnvol;
										fi;
									fi;
								od;
								if tB<>mintB then
									tB:=mintB;
									tBinv:=InverseMat(tB);
								fi;
							fi;
						fi;
					od;
					if not tB in doneBs then
						pivotcounter:=pivotcounter+1;
						Add(doneBs, tB);
						tBinv:=InverseMat(tB);
					else
						pivotcounter:=infinity;
					fi;
					relevants:=newttM;
				od;
			fi;
			tt:=tt+1;
		od;
		return(fail);
	end;
	#
	#
	########################
	#
	counter:=0;
	isawallflag:=false;
	while counter<trial*n do
		#Printn("go findtrue");
		res:=findtrue();
		if res<>fail then
			return([true, res, irrelevants]);
		fi;
		#Printn(counter, "after findtrue", Length(onedimfaces), Length(irrelevants));
		findfalsetrial :=findfalsetrial +n*trial;
		if onedimfaces=[] then
			#Printn("go findfalse");
			res:=findfalse(false);
			if res<>fail then
				if n=Length(linforms) then
					 beep(355456);
				fi;
				return([false, res]);
			fi;
		else
			res:=findfalse(true);
			if res<>fail then
				if n=Length(linforms) then
					 beep(35545116);
				fi;
				return([false, res]);
			fi;
			findtruetrial:=findtruetrial+n*trial;
		fi;
		#Printn(counter, "after findfalse", Length(onedimfaces), Length(irrelevants));
		counter:=counter+1;
	od;
	#
	########################
	#
	Printn("cannot Determine RandFindInteriorPoint"); 
	Error();
	return(fail);
end;


IntPointCheckcounter:=0;
IntPointCheck:=function(linforms, result)
	local tv, tB, vv, xx, signs, beep,
	relevantlinforms, irdata, tlf, n;
	#
	beep:=function(numb)
		Printn("IntPointCheck", numb); Error();
	end;
	#
	n:=Length(linforms[1]);
	if result[1] then 
		signs:=Set(linforms, tv->SignRat(result[2]*tv));
		if signs<>[1] then
			beep(82233);
		fi;
		#
		if n>2 then 
			relevantlinforms:=SetMinus(linforms, List(result[3], xx->xx[1]));
			for irdata in result[3] do # result[3] is IRrerevant
				tlf:=irdata[1];
				vv:=irdata[2];
				tB:=irdata[3];
				if tlf<>vv*tB then
					beep(114561);
				fi;
				if not IsSubset(linforms, tB) then
					beep(45611);
				fi;
				if not tlf in linforms then
					beep(533411);
				fi;
				if -1 in Set(vv, SignRat) then
					beep(99112211);
				fi;
				if IsAWall(relevantlinforms, tlf, 1)[1]<>false then
					beep(5881176);
				fi;
			od;
		fi;
	else
		tlf:=result[2][1];
		vv:=result[2][2];
		tB:=result[2][3];
		if tlf<>vv*tB then
			beep(11456);
		fi;
		if not IsSubset(linforms, tB) then
			beep(456);
		fi;
		if not tlf in linforms then
			beep(5334);
		fi;
		if 1 in Set(vv, SignRat) then
			beep(992211);
		fi;
	fi;
	IntPointCheckcounter:=IntPointCheckcounter+1;
	#Printn("intptcheckcounter", IntPointCheckcounter, result[1]);
	return(true);
end;
