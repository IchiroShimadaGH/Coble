FindNameGlobal:=function(obj)
	local names, name;
	names:=NamesUserGVars( );
	for name in names do
		if IsIdenticalObj(obj, ValueGlobal(name)) then
			return(name);
		fi;
	od;
	Print("In FindNameGlobal (SavesReads.txt): No name in NamesUserGVars for ", obj, ";\n");
	Error();
end;

fixsavedata:=function(obj)
	local fn,name;
	MakeImmutable(obj);
	name:=FindNameGlobal(obj);
	MakeReadOnlyGlobal(name);
	fn:=Filename(Directory("./gapsaves"), Concatenation("save", name, ".txt"));
	PrintTo(fn, name, ":=", obj, ";\n");
	AppendTo(fn, "MakeImmutable(", name, ");\n");
	AppendTo(fn, "MakeReadOnlyGlobal(\"", name, "\");\n");
	Printn("fix and save:", name);
end;

savedataver:=function(obj, versionnumb)
	local fn,name;
	if not (IsInt(versionnumb) and versionnumb>=0) then
		Printn("wrong versionnumb");Error();
	fi;
	name:=FindNameGlobal(obj);
	name:=Concatenation(name, "_Ver", String(versionnumb));
	fn:=Filename(Directory("./gapsaves"), Concatenation("save", name, ".txt"));
	if IsExistingFile(fn)  then
		Printn("existing versionnumb");Error();
	fi;
	PrintTo(fn, name, ":=", obj, ";\n");
end;
