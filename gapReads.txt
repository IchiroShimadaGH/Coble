SetRecordFiles:=function()
	#
	# change on 2025/04/27
	#
	local date, path, str, a, L, aa;
	path := DirectoriesSystemPrograms();
	date := Filename( path, "date" );
	str := "";
	a := OutputTextString(str,true);
	Process( DirectoryCurrent(), date, InputTextNone(), a, ["+%Y-%m-%d %H:%M:%S"] );
	L:=[];
	for aa in str do
		if IsDigitChar(aa) then Add(L, aa); fi;
	od;
	CloseStream(a);
	sessionnumb:= EvalString(L);
	#wspfn:=Concatenation(["gapwsps/gapwsp", L, ".gwsp"]);
	#SaveOnExitFile:=wspfn;
	if IsBoundGlobal("SaveOnExitFile") then
		UnbindGlobal("SaveOnExitFile");
	fi;
	logfn:=Concatenation(["gaplogs/gaplog", L, ".txt"]);
	LogTo(logfn);
end;

SetRecordFiles();
